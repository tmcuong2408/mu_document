	\begin{abstract}
		In modern computational science and engineering, handling uncertain data is a core challenge. Traditional tools such as Classical Interval Arithmetic often suffer from the severe "dependency problem" --- where the repetition of algebraic variables generates endless "arithmetic noise" and causes explosive expansion of computational bounds. Similarly, Probabilistic or Fuzzy Set methods demand strict prior information and incur extremely high computational costs when scenario dimensions swell.
		
		This work proposes a comprehensive mathematical paradigm entitled Extended Logic and Mathematics of Uncertainty to thoroughly resolve these bottlenecks:
		\begin{itemize}[leftmargin=*]
			\item Canonical Form \& Lazy Evaluation Mechanism: Representing uncertain numbers via generating functions on index domains. This structure preserves the history of algebraic correlations to eliminate arithmetic noise at its root, while employing a lazy evaluation mechanism to represent and store infinite-element uncertain numbers without consuming RAM, performing actual calculations only upon evaluation requests.
			
			\item Four Arithmetic Spaces System:
			\begin{itemize}[leftmargin=*]
				\item Single-point Space $(\circ)_1$ \& Extended Single-point Space $(\circ)_{1'}$: Preserving algebraic consistency, internal correlation, and expanding the ability to process inverse operations/hierarchical structures.
				\item Minkowski Space $(\circ)_m$ \& Extended Minkowski Space $(\circ)_{m'}$: Modeling interactions, independent noise propagation across uncertain sources, and expanding operators for solving arithmetic equations.
			\end{itemize}
			
			\item Uncertain Arithmetic, Algebra \& Calculus:
			\begin{itemize}[leftmargin=*]
				\item Uncertain Arithmetic \& Algebra: Constructing binary operations, weak relations, weights, uncertain algebraic equations, and compressing complex intersection/union/element-checking operations into root-finding problems on index domains with $\mathcal{O}(1)$ complexity.
				\item Uncertain Calculus: Fully building the theories of Limits, Continuity, Difference Quotients, Derivatives, and Uncertain Integrals over scenario sets.
			\end{itemize}
		\end{itemize}
		
		This work opens up a powerful computational framework serving directly for problems in noisy data processing, simulation of uncertain systems, inverse problems in geophysics, and optimal decision-making.
		
		\paragraph{Keywords:} Uncertain arithmetic, Uncertain algebra, Canonical form, Lazy evaluation, Single-point space, Minkowski space, Extended Minkowski space, Uncertain calculus, Arithmetic noise elimination.
	\end{abstract}

	\chapter{Extended Logic}
	
	\section{State Space and Atomic Propositions}
	Let $(\Omega, \mathcal{F}, \lambda)$ be a defined measure space representing the set of possible scenarios or states of the system.
	
	\begin{definition}[Atomic Proposition]
		An atomic proposition $P$ in the Extended Logic system is defined alongside a scenario truth function $v(P, \cdot) : \Omega \to \{0, 1\}$, where:
		\[
		v(P, \omega) = \begin{cases} 
			1 & \text{if } P \text{ is true at state } \omega, \\
			0 & \text{if } P \text{ is false at state } \omega.
		\end{cases}
		\]
	\end{definition}
	
	\begin{definition}[Truth Valuation Operator]
		The truth valuation operator $\ddclmd$ is a mapping from the set of propositions to the interval $[0, 1]$, defined by the measure of the set of states that satisfy the proposition:
		\[
		\ddclmd(P) := \frac{\lambda(\{\omega \in \Omega : v(P, \omega) = 1\})}{\lambda(\Omega)}
		\]
	\end{definition}
	
	\section{Algebra of Compound Propositions and Logical Connectives}
	Classical logical connectives (Negation $\neg$, Conjunction $\land$, Disjunction $\lor$, Implication $\Rightarrow$) are extended to the state space via Boolean algebra pointwise for each $\omega \in \Omega$.
	
	\begin{definition}[Valuation of Compound Propositions]
		Let $P_1, P_2, \dots, P_n$ be atomic propositions and $\Phi$ be a compound proposition constructed from classical logical connectives. The truth value at state $\omega \in \Omega$ of $\Phi$ is determined by:
		\[
		v(\Phi(P_1, \dots, P_n), \omega) = \Phi\big(v(P_1, \omega), v(P_2, \omega), \dots, v(P_n, \omega)\big) \in \{0, 1\}
		\]
		Then, the global truth value $\ddclmd(\Phi) \in [0, 1]$ is computed by:
		\[
		\ddclmd(\Phi) := \frac{\lambda(\{\omega \in \Omega : v(\Phi(P_1, \dots, P_n), \omega) = 1\})}{\lambda(\Omega)}
		\]
		and we denote the proposition accompanied by its truth measure as $(\Phi)_{V(\Phi)}$.
	\end{definition}
	
	\begin{theorem}[Preservation of Classical Logical Tautologies]
		If $\Phi$ is a tautology of Classical Propositional Logic, then $\ddclmd(\Phi) = 1$. In particular, fundamental laws are always absolutely preserved:
		\begin{enumerate}[leftmargin=*]
			\item Law of non-contradiction: $\ddclmd(P \land \neg P) = 0$.
			\item Law of excluded middle: $\ddclmd(P \lor \neg P) = 1$.
		\end{enumerate}
	\end{theorem}
	
	\begin{proof}
		Since $\Phi$ is a classical tautology, for every combination of binary values of atomic propositions, we always have $\Phi\big(v(P_1, \omega), \dots, v(P_n, \omega)\big) = 1$ for all $\omega \in \Omega$.
		Therefore, $\{\omega \in \Omega : v(\Phi, \omega) = 1\} = \Omega$. It follows that:
		\[
		\ddclmd(\Phi) = \frac{\lambda(\Omega)}{\lambda(\Omega)} = 1
		\]
	\end{proof}
	
	\section{Implication Relation and Inference Rules}
	
	\begin{definition}[Implication and Degree of Inference]
		Let $P$ and $Q$ be two propositions. The implication $P \Rightarrow Q$ at state $\omega$ is defined according to classical logic as $v(P \Rightarrow Q, \omega) = v(\neg P \lor Q, \omega)$. The global truth value of the implication is:
		\[
		\ddclmd(P \Rightarrow Q) = \frac{\lambda(\{\omega \in \Omega : v(P, \omega) \le v(Q, \omega)\})}{\lambda(\Omega)}
		\]
	\end{definition}
	
	\begin{theorem}[Extended Modus Ponens Lemma]
		For any two propositions $P$ and $Q$, the following inference inequality always holds:
		\[
		\ddclmd(Q) \ge \ddclmd(P) + \ddclmd(P \Rightarrow Q) - 1
		\]
	\end{theorem}
	\begin{proof}
		Consider the state space $(\Omega, \mathcal{F}, \lambda)$ and let $P, Q$ be two arbitrary propositions. 
		
		For each state $\omega \in \Omega$, let $v(P, \omega) \in \{0, 1\}$ and $v(Q, \omega) \in \{0, 1\}$ be the scenario truth values of $P$ and $Q$, respectively. We define the following state subsets of $\Omega$:
		\begin{align*}
			\Omega_P &= \{\omega \in \Omega : v(P, \omega) = 1\}, \\
			\Omega_Q &= \{\omega \in \Omega : v(Q, \omega) = 1\}, \\
			\Omega_{P \Rightarrow Q} &= \{\omega \in \Omega : v(P \Rightarrow Q, \omega) = 1\}.
		\end{align*}
		
		By the definition of classical material implication on state space, for any $\omega \in \Omega$, $v(P \Rightarrow Q, \omega) = 1$ if and only if $v(P, \omega) \le v(Q, \omega)$. Thus:
		\[
		\Omega_{P \Rightarrow Q} = \{\omega \in \Omega : v(P, \omega) \le v(Q, \omega)\}
		\]
		
		We evaluate the truth value at state $\omega$ for each case:
		\begin{enumerate}[leftmargin=*]
			\item If $\omega \in \Omega_P$ (so $v(P, \omega) = 1$), then $v(P, \omega) \le v(Q, \omega)$ holds if and only if $v(Q, \omega) = 1$ (so $\omega \in \Omega_Q$). Therefore:
			\[
			\Omega_P \cap \Omega_{P \Rightarrow Q} = \Omega_P \cap \Omega_Q
			\]
			\item Since $\Omega_P \cap \Omega_Q \subseteq \Omega_Q$, applying the monotonicity property of measure $\lambda$ yields:
			\[
			\lambda(\Omega_P \cap \Omega_{P \Rightarrow Q}) = \lambda(\Omega_P \cap \Omega_Q) \le \lambda(\Omega_Q)
			\]
		\end{enumerate}
		
		On the other hand, for any two measurable subsets $\Omega_P$ and $\Omega_{P \Rightarrow Q}$ of $\Omega$, by the principle of inclusion-exclusion (or Bonferroni's inequality for measures), we have:
		\[
		\lambda(\Omega_P \cup \Omega_{P \Rightarrow Q}) = \lambda(\Omega_P) + \lambda(\Omega_{P \Rightarrow Q}) - \lambda(\Omega_P \cap \Omega_{P \Rightarrow Q})
		\]
		Since $\Omega_P \cup \Omega_{P \Rightarrow Q} \subseteq \Omega$, its measure is bounded above by $\lambda(\Omega)$:
		\[
		\lambda(\Omega_P) + \lambda(\Omega_{P \Rightarrow Q}) - \lambda(\Omega_P \cap \Omega_{P \Rightarrow Q}) \le \lambda(\Omega)
		\]
		Rearranging the inequality gives:
		\[
		\lambda(\Omega_P \cap \Omega_{P \Rightarrow Q}) \ge \lambda(\Omega_P) + \lambda(\Omega_{P \Rightarrow Q}) - \lambda(\Omega)
		\]
		
		Combining this with $\lambda(\Omega_Q) \ge \lambda(\Omega_P \cap \Omega_{P \Rightarrow Q})$, we obtain:
		\[
		\lambda(\Omega_Q) \ge \lambda(\Omega_P) + \lambda(\Omega_{P \Rightarrow Q}) - \lambda(\Omega)
		\]
		
		Dividing both sides of the inequality by the total measure $\lambda(\Omega) > 0$:
		\[
		\frac{\lambda(\Omega_Q)}{\lambda(\Omega)} \ge \frac{\lambda(\Omega_P)}{\lambda(\Omega)} + \frac{\lambda(\Omega_{P \Rightarrow Q})}{\lambda(\Omega)} - 1
		\]
		
		By the definition of the global truth valuation operator $\ddclmd$, we conclude:
		\[
		\ddclmd(Q) \ge \ddclmd(P) + \ddclmd(P \Rightarrow Q) - 1
		\]
		This completes the proof.
	\end{proof}
	\section{Illustrative Example of Correlation Preservation}
	\begin{example}
		Consider state space $\Omega = \{1, 2, 3, 4\}$ with counting measure $\lambda(A) = |A|$. Consider two atomic propositions $P$ and $Q$ having scenario truth functions:
		\begin{itemize}[leftmargin=*]
			\item $P$ is true at $\Omega_P = \{1, 2\} \implies \ddclmd(P) = 2/4 = 0.5$.
			\item $Q$ is true at $\Omega_Q = \{2, 3\} \implies \ddclmd(Q) = 2/4 = 0.5$.
		\end{itemize}
		The set of states that makes the compound proposition $P \land Q$ true is $\Omega_P \cap \Omega_Q = \{2\}$. Therefore:
		\[
		\ddclmd(P \land Q) = \frac{|\{2\}|}{4} = 0.25
		\]
		\textit{Remark:} If using the classical Fuzzy operator $\min(\ddclmd(P), \ddclmd(Q)) = \min(0.5, 0.5) = 0.5$, the result would be distorted due to ignoring scenario shifts between $P$ and $Q$. Direct valuation on state space $\Omega$ preserves the essence of logical correlation.
	\end{example}

	\section{Introduction}
	In daily life, we frequently encounter problems where input data is not a deterministic single value, but is only known to belong to some set of possible scenarios. We consider the following two introductory examples:
	
	\begin{example}
		A coffee shop $A$ serves a daily customer count belonging to set $K = \set{10, 15, 30}$ (customers). The average spending per customer belongs to set $C = \set{10, 20, 50}$ (thousand VND). The set of all possible daily revenue levels for coffee shop $A$ is determined by:
		\[
		D = \set{kc : k \in K, \, c \in C}  \text{ (thousand VND)}
		\]
	\end{example}
	
	\begin{example}
		Mr. $A$ plans to travel from Hanoi to Ho Chi Minh City via 2 legs:
		\begin{itemize}[leftmargin=*]
			\item Leg $A_1$: Possible distance $S_1 \in \set{200, 240, 300}$ (km), travel speed $V_1 \in \set{40, 50}$ (km/h).
			\item Leg $A_2$: Possible distance $S_2 \in \set{400, 500, 600}$ (km), travel speed $V_2 \in \set{80, 100, 120}$ (km/h).
		\end{itemize}
		The possible travel times on each leg are given respectively by the sets:
		\[
		T_1 = \set{\frac{s_1}{v_1} : s_1 \in S_1, \, v_1 \in V_1} \text{ (hours)}
		\]
		\[
		T_2 = \set{\frac{s_2}{v_2} : s_2 \in S_2, \, v_2 \in V_2}  \text{ (hours)}
		\]
		Thus, the total possible travel time for the whole journey belongs to the set:
		\[
		T = \set{t_1 + t_2 : t_1 \in T_1, \, t_2 \in T_2} \text{ (hours)}
		\]
		From this, we can derive characteristic laws of the result set $T$, such as minimum value $\min(T)$, maximum value $\max(T)$, mean of $T$, or determine whether $T$ satisfies a certain condition function or criterion $f(T)$. Grasping this complete structural range forms the core foundation for solving decision-making problems under data uncertainty.
	\end{example}
	
	To automate and systemize direct computations over these sets of possible scenarios, we construct Extended Logic and Mathematics of Uncertainty — an extension tool of classical arithmetic for quantities exhibiting uncertainty.
	
	\section{Uncertain Number}
	\begin{definition}[Uncertain Number over Field $\mathbb{K}$]
		Let $\mathbb{K}$ be a field; an uncertain number $A$ is a quantity whose value belongs to a set $A \subseteq \mathbb{K}$. The set of all uncertain numbers over field $\mathbb{K}$ is denoted by $\tsbd{K}$. If $A =\s{a}$, we identify $A$ with element $a$. If $A =\emptyset$, we say $A$ is undefined. The number $\emptyset \in \tsbd{K}$ is called the null number.
	\end{definition}
	
	\begin{definition}[Transfer Operator]
		To resolve notation ambiguity between sets and uncertain numbers in certain cases, we define the following two transfer operators:
		\begin{enumerate}[leftmargin=*]
			\item Given $A \in \tsbd{K}$, we denote $A_s$ as the set containing all possible values of $A$.
			\item Given $A$ as a set, the operator transforming $A$ into an uncertain number is denoted by $A_u \in \tsbd{K}$.
		\end{enumerate}
	\end{definition}
	
	\begin{remark}
		In later sections of this document, we will prove $f(A_s) \neq f(A)$ when $A$ is an uncertain number. This is why uncertain numbers are denoted by $\{\circ\}_u$ rather than standard set notation.
	\end{remark}
	
	\begin{remark}
		Throughout this document, we adopt the conventions:
		\begin{enumerate}[leftmargin=*]
			\item Single-valued numbers are denoted by lowercase letters such as $x, y, \dots$.
			\item Uncertain numbers are denoted by uppercase letters such as $X, Y, \dots$.
			\item Sets of uncertain numbers are denoted by calligraphic uppercase letters such as $\mathcal{X}, \mathcal{Y}, \dots$.
			\item Arithmetic spaces are notation conventions to avoid constructing multiple distinct symbols for the same operation within each arithmetic space. The following is the list of arithmetic space notations:
			\begin{enumerate}[leftmargin=*]
				\item $\kgdd{\circ}$: single-point arithmetic space.
				\item $\kgddmr{\circ}$: extended single-point arithmetic space.
				\item $\kgdt{\circ}$: Minkowski arithmetic space.
				\item $\kgdtmr{\circ}$: extended Minkowski arithmetic space.
			\end{enumerate}
		\end{enumerate}
	\end{remark}
	
	\begin{definition}[Dimension]
		The finite dimension of a finite-element uncertain number $A \in \tsbd{K}$ equals the number of elements in $A$, denoted by $\dimU(A)$. If $A$ contains infinitely many elements, we say the dimension of $A$ is infinite, written as $\dimU(A) = \infty$.
	\end{definition}
	
	\begin{definition}[Canonical Form]
		Let $A \in \tsbd{K}$. We call $f_A : d_A \to \mathbb{K}$ a generating function of uncertain number $A$ if there exists an index set $d_A \subseteq \mathbb{K}$ satisfying:
		\[d_A = \begin{cases*}
			\set{1..n} \subseteq \mathbb{N^*} \text{ if } \dimU(A)=n \\
			\mathbb{N^*} \text{ if } A_s \text{ is a countably infinite set  }\\
			[0,1] \text{ if } A_s \text{ is an uncountably infinite set.}
		\end{cases*}\]
		such that $\bd{f_A(d_A)}= A$. In this case, we call $(f_A,d_A)$ a canonical form of $A$ and denote $A \sim (f_A,d_A)$.
	\end{definition}
	
	\begin{theorem}[Equivalence Relation of Canonical Forms]
		\label{thm:canonical_form_equivalence}
		Let $\mathbf{CF}(A)$ be the set of all canonical forms of an uncertain number $A \in \tsbd{K}$. Define a binary relation $\sim$ on $\mathbf{CF}(A)$ such that two canonical forms $(f_1, d_1)$ and $(f_2, d_2)$ are equivalent, written as $(f_1, d_1) \sim (f_2, d_2)$, if and only if:
		\begin{enumerate}[leftmargin=*]
			\item They share the same index domain: $d_1 = d_2 = d_A$.
			\item There exists a bijection $\sigma : d_A \to d_A$ on the index domain such that $f_2 = f_1 \circ \sigma$.
		\end{enumerate}
		Then, the relation $\sim$ is an equivalence relation on $\mathbf{CF}(A)$.
	\end{theorem}
	
	\begin{proof}
		To establish that $\sim$ is an equivalence relation on $\mathbf{CF}(A)$, we verify the three defining properties: reflexivity, symmetry, and transitivity.
		
		\begin{enumerate}[leftmargin=*]
			\item Reflexivity: 
			Let $(f_1, d_A) \in \mathbf{CF}(A)$. Consider the identity mapping $\sigma = \text{id}_{d_A} : d_A \to d_A$ defined by $\sigma(t) = t$ for all $t \in d_A$. Obviously, $\text{id}_{d_A}$ is a bijection on $d_A$, and we have:
			\[
			(f_1 \circ \text{id}_{d_A})(t) = f_1(t), \quad \forall t \in d_A \implies f_1 = f_1 \circ \text{id}_{d_A}
			\]
			Hence, $(f_1, d_A) \sim (f_1, d_A)$.
			
			\item Symmetry: 
			Let $(f_1, d_A), (f_2, d_A) \in \mathbf{CF}(A)$ such that $(f_1, d_A) \sim (f_2, d_A)$. By definition, there exists a bijection $\sigma : d_A \to d_A$ satisfying $f_2 = f_1 \circ \sigma$.
			
			Since $\sigma$ is a bijection, its inverse mapping $\sigma^{-1} : d_A \to d_A$ exists and is also a bijection on $d_A$. Composing both sides with $\sigma^{-1}$ yields:
			\[
			f_2 \circ \sigma^{-1} = (f_1 \circ \sigma) \circ \sigma^{-1} = f_1 \circ (\sigma \circ \sigma^{-1}) = f_1 \circ \text{id}_{d_A} = f_1
			\]
			Thus, $f_1 = f_2 \circ \sigma^{-1}$ with bijection $\sigma^{-1}$, which implies $(f_2, d_A) \sim (f_1, d_A)$.
			
			\item Transitivity: 
			Let $(f_1, d_A), (f_2, d_A), (f_3, d_A) \in \mathbf{CF}(A)$ such that $(f_1, d_A) \sim (f_2, d_A)$ and $(f_2, d_A) \sim (f_3, d_A)$. 
			
			By definition, there exist bijections $\sigma_1 : d_A \to d_A$ and $\sigma_2 : d_A \to d_A$ such that:
			\[
			f_2 = f_1 \circ \sigma_1 \quad \text{and} \quad f_3 = f_2 \circ \sigma_2
			\]
			Substituting $f_2$ into the expression for $f_3$:
			\[
			f_3 = (f_1 \circ \sigma_1) \circ \sigma_2 = f_1 \circ (\sigma_1 \circ \sigma_2)
			\]
			Since the composition of two bijections on $d_A$ is again a bijection on $d_A$, the mapping $\sigma_3 = \sigma_1 \circ \sigma_2 : d_A \to d_A$ is a bijection satisfying $f_3 = f_1 \circ \sigma_3$. Consequently, $(f_1, d_A) \sim (f_3, d_A)$.
		\end{enumerate}
		
		Since $\sim$ is reflexive, symmetric, and transitive, it constitutes a well-defined equivalence relation on $\mathbf{CF}(A)$.
	\end{proof}
	
	\begin{remark}
		The equivalence relation $\sim$ partitions the space of all generating representations of an uncertain number $A$ into equivalence classes $[\,(f_A, d_A)\,]_\sim$. Each equivalence class corresponds to the unique invariant algebraic identity of $A$ up to scenario reparameterization, ensuring that all physical/algebraic properties of $A$ remain invariant under index permutation.
	\end{remark}

	\begin{definition}[Distance]

	\begin{definition}[Weak Binary Relation]
		Consider $\mathcal{R}$ as a binary relation on field $\mathbb{K}$ and two uncertain numbers $A \sim (f_A, d_A)$, $B \sim (f_B, d_B)$ in $\mathbf{U}(\mathbb{K})$. 
		
		The weak binary relation $A \mathcal{R} B$ is a proposition evaluated on scenario space $\Omega = d_A \times d_B$ with atomic truth function:
		\[
		v(A \mathcal{R} B, \omega) = \begin{cases} 
			1 & \text{if } f_A(t_1) \mathcal{R} f_B(t_2) \text{ is true}, \\
			0 & \text{if } f_A(t_1) \mathcal{R} f_B(t_2) \text{ is false},
		\end{cases} \quad \text{with } \omega = (t_1, t_2) \in \Omega
		\]
		Then, the truth value of relation $A \mathcal{R} B$ is determined by:
		\[
		\ddclmd(A \mathcal{R} B) := \frac{\lambda\big(\{(t_1, t_2) \in d_A \times d_B : f_A(t_1) \mathcal{R} f_B(t_2)\}\big)}{\lambda(d_A \times d_B)}
		\]
	\end{definition}
	
	\begin{example}
		Given $A = \s{1,2}$ as an uncertain number, we have $$\gtcl{A\leq A}{0.75}$$ $$\gtcl{A>A}{0.25}$$
		We remark: $\ddclmd(A\leq A) + \ddclmd(A>A) = 1$. Although $A=A$, their sizes differ probabilistically/scenariowise.
	\end{example}
	
	\begin{example}
		Given $A = \s{1,2}$ and $B = \s{3} = 3$, then $A \leq B$.
	\end{example}
	
	\begin{theorem}[Balance]
		\label{thm:balance}
		Let $X \in \tsbd{R}$ be an uncertain number with canonical form $X \sim (f_X, d_X)$. Assume that the index domain $d_X \subseteq \mathbb{R}$ is equipped with a non-atomic, finite measure $\lambda$ (such as the standard Lebesgue measure on $[0, 1]$), and the generating function $f_X: d_X \to \mathbb{R}$ is measurable. 
		
		If the level set of diagonals $\mathcal{D} = \{(t_1, t_2) \in d_X \times d_X : f_X(t_1) = f_X(t_2)\}$ has product measure zero, i.e., $(\lambda \times \lambda)(\mathcal{D}) = 0$, then the truth valuation of the weak inequality $X \leq X$ satisfies:
		\[
		\ddclmd(X \leq X) = \frac{1}{2}
		\]
	\end{theorem}
	
	\begin{proof}
		Consider the product measure space $(\Omega, \mathcal{F}_{\Omega}, \mu)$, where $\Omega = d_X \times d_X$ represents the scenario space of the weak binary relation $X \leq X$, $\mathcal{F}_{\Omega} = \mathcal{F} \otimes \mathcal{F}$, and $\mu = \lambda \times \lambda$ is the product measure.
		
		According to the definition of weak binary relation, the truth measure of $X \leq X$ is given by:
		\[
		\ddclmd(X \leq X) = \frac{\mu(E_{\leq})}{\mu(\Omega)}
		\]
		where $E_{\leq} = \{(t_1, t_2) \in d_X \times d_X : f_X(t_1) \leq f_X(t_2)\}$.
		
		We decompose the scenario space $\Omega = d_X \times d_X$ into three mutually disjoint measurable subsets:
		\begin{align*}
			E_{<} &= \{(t_1, t_2) \in d_X \times d_X : f_X(t_1) < f_X(t_2)\}, \\
			E_{>} &= \{(t_1, t_2) \in d_X \times d_X : f_X(t_1) > f_X(t_2)\}, \\
			\mathcal{D} &= \{(t_1, t_2) \in d_X \times d_X : f_X(t_1) = f_X(t_2)\}.
		\end{align*}
		Thus, $\Omega = E_{<} \cup E_{>} \cup \mathcal{D}$ and $\mu(\Omega) = \mu(E_{<}) + \mu(E_{>}) + \mu(\mathcal{D})$.
		
		By the swap mapping $S: (t_1, t_2) \mapsto (t_2, t_1)$, which is measure-preserving on the product space $(\Omega, \mu)$, we observe that $S(E_{<}) = E_{>}$. By the measure-preserving property of Fubini's theorem / product measures:
		\[
		\mu(E_{<}) = \mu(E_{>})
		\]
		By assumption, $(\lambda \times \lambda)(\mathcal{D}) = 0$. Therefore:
		\[
		\mu(\Omega) = \mu(E_{<}) + \mu(E_{>}) = 2 \cdot \mu(E_{<}) \implies \frac{\mu(E_{<})}{\mu(\Omega)} = \frac{1}{2}
		\]
		Since $E_{\leq} = E_{<} \cup \mathcal{D}$ and $\mu(\mathcal{D}) = 0$, we have:
		\[
		\mu(E_{\leq}) = \mu(E_{<}) + \mu(\mathcal{D}) = \mu(E_{<})
		\]
		Consequently,
		\[
		\ddclmd(X \leq X) = \frac{\mu(E_{\leq})}{\mu(\Omega)} = \frac{\mu(E_{<})}{\mu(\Omega)} = \frac{1}{2}
		\]
		This completes the proof.
	\end{proof}
	
	\begin{definition}[Set Arithmetic Operations]
		Let $*$ be an arbitrary set operation, $A, B \in \mathbf{U}(\mathbb{K})$, we define:
		\[
		A * B := (A_s * B_s)_u
		\]
		Where $A_s * B_s$ on the right hand side represents the set operation.
	\end{definition}
	
	\begin{example}
		Consider the problem: Person $A$ has an uncertain number of apples $A=\s{1,2}$. $A$ intends to share these apples with room $B$, where room $B$ has an uncertain number of people $B = \s{1,0}$. After $A$ divides the apples, each person in room $B$ receives a quantity of apples equal to 
		\[\frac{A}{B} = \s{1,2}\]
		Why does $\frac{1}{0}$ not appear in the result set? Because if $A$ sees no people in the room, $A$ does not perform division, nor can $A$ divide apples to each person in room $B$ with $+\infty$ apples. This division does not happen rather than yielding no valid outcome. Hence we define binary operations on the set of uncertain numbers as follows:
	\end{example}
	
	\begin{definition}[Single-Point Arithmetic Space]
		Let $A \in \tsbd{K}$ be an uncertain set. We define a single-point uncertain function as a mapping $f$ acting on variable $A$, such that during each evaluation pass, variable $A$ receives a unique representative point value $x \in A$ at all occurrences of $A$ in expression $f$.
		
		The image set of the corresponding point-wise uncertain function, denoted $\kgdd{f}$, is determined by:
		\[
		\kgdd{f}(A) := \s{f(x) \;:\; x \in A}
		\]
		Operations within $(\circ)_1$ are called operations on the single-point arithmetic space.
	\end{definition}
	
	\begin{example}
		Let $f : \mathcal{X} \to \mathcal{Y}$ be given by:
		\[f(X) = (X^2+5X+6)_1\] as a single-point uncertain function.
	\end{example}
	
	\begin{remark}
		This definition ensures algebraic consistency via two core mechanisms:
		\begin{itemize}[leftmargin=*]
			\item Internal Point Identity: Every symbol of variable $A$ appearing in expression $f$ (regardless of position or operation) is bound to receive the same value $x \in A$ during the same mapping pass.
			\item Elimination of Quantity Expansion: Thanks to unified point assignment, classical algebraic properties are fully preserved, such as $\kgdd{x - x}(A) = \s{0}$ and $\kgdd{x + x}(A) = 2A$.
		\end{itemize}
	\end{remark}
	
	\begin{example}
		Consider single-variable function $f(A) = A^2 - 3A + 2$ with $A \in \tsbd{\mathbb{R}}$. The image set of this single-point uncertain function is defined by:
		\[
		\kgdd{f}(A) = \s{x^2 - 3x + 2 \;:\; x \in A}
		\]
		During each evaluation $x \in A$, both positions $A^2$ and $-3A$ uniformly receive the same value $x$.
	\end{example}
	
	\begin{definition}[Minkowski Arithmetic Space]
		Let $\mathbb{K}$ be a field, with * as any binary operation on $\mathbb{K}$, we define arithmetic binary operation * on $\tsbd{K}$:
		For $A,B \in \tsbd{K}$, then 
		$$(A * B)_m = \s{a*b : a \in A , b\in B , a*b \in \mathbb{K}}$$
		If $A * B =\emptyset$, we say $A * B$ is undefined. In writing, we abbreviate $(A*B)_m$ as $A*B$ and refer to operations in $(\circ)_m$ as operations executed on Minkowski arithmetic space.
	\end{definition}
	
	\begin{definition}[Extended Minkowski Arithmetic Space]
		Let $\mathbb{K}$ be a field. The Extended Minkowski Arithmetic Space $(\circ)_{m'}$ is an extension of the Minkowski arithmetic space $(\circ)_m$ defined to incorporate inverse operations, fractional coefficient scaling, and solve operator equations over uncertain quantities. 
		
		For any $A, B \in \tsbd{K}$ and binary operation $*$, operations in $(\circ)_{m'}$ admit extended algebraic solutions such that higher-order uncertain structures, equations, and roots (such as $X^q_* = A^p_*$) are fully closed and well-defined.
	\end{definition}

	\begin{theorem}[Canonical Form on Minkowski Arithmetic Space]
		\label{thm:minkowski_canonical_form}
		Let $A, B \in \tsbd{K}$ be two uncertain numbers with canonical forms $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$, respectively. Let $*$ be any binary operation on field $\mathbb{K}$ extended to the Minkowski arithmetic space $(\circ)_m$.
		
		Then, the resulting uncertain number $C = \kgdt{A * B}$ possesses a canonical form $(f_{A*B}, d_{A*B})$ uniquely determined by:
		\begin{enumerate}[leftmargin=*]
			\item Scenario Index Domain: 
			\[
			d_{A*B} = d_A \times d_B
			\]
			\item Generating Function: $f_{A*B} : d_A \times d_B \to \mathbb{K}$ defined by
			\[
			f_{A*B}(x_1, x_2) = f_A(x_1) * f_B(x_2), \quad \forall (x_1, x_2) \in d_A \times d_B
			\]
		\end{enumerate}
		Consequently, the set-theoretic representation of the resulting uncertain number is:
		\[
		\kgdt{A * B} = \bd{f_{A*B}(d_A \times d_B)} = \s{f_A(x_1) * f_B(x_2) \;:\; (x_1, x_2) \in d_A \times d_B}
		\]
	\end{theorem}
	
	\begin{proof}
		To establish that $(f_{A*B}, d_A \times d_B)$ is a valid canonical form of $\kgdt{A * B}$, we must verify two key criteria: index domain compatibility and set-image preservation under the generating function.
		
		\paragraph{Step 1: Index Domain Compatibility.}
		According to the definition of canonical form, the scenario index domain of an uncertain number corresponds to the cardinality of its underlying set of scenario values.
		\begin{itemize}[leftmargin=*]
			\item Finite Case: If $\dimU(A) = n_1$ and $\dimU(B) = n_2$, then $d_A = \{1, \dots, n_1\}$ and $d_B = \{1, \dots, n_2\}$. The Cartesian product $d_A \times d_B$ contains $n_1 \cdot n_2$ element pairs, isomorphic to $\{1, \dots, n_1 \cdot n_2\}$ under a lexicographical bijection.
			\item Countably Infinite Case: If $d_A, d_B \subseteq \mathbb{N}^*$, since the Cartesian product of two countable sets is countable, we have $d_A \times d_B \sim \mathbb{N}^*$.
			\item Uncountably Infinite Case: If $d_A = [0, 1]$ and $d_B = [0, 1]$, the product domain $d_A \times d_B = [0, 1]^2$ exhibits the cardinality of the continuum, which is isomorphic to $[0, 1]$ via standard space-filling bijections.
		\end{itemize}
		Thus, $d_{A*B} = d_A \times d_B$ is a well-defined scenario domain representing independent uncertainty interactions.
		
		\paragraph{Step 2: Set-Image Preservation.}
		We show that $f_{A*B}(d_A \times d_B) = (\kgdt{A * B})_s$ by establishing mutual set inclusions:
		
		\begin{enumerate}[leftmargin=*]
			\item Forward Inclusion ($\subseteq$): 
			Let $y \in f_{A*B}(d_A \times d_B)$. By definition of the generating function, there exists a scenario index pair $(x_1, x_2) \in d_A \times d_B$ such that:
			\[
			y = f_{A*B}(x_1, x_2) = f_A(x_1) * f_B(x_2)
			\]
			Since $x_1 \in d_A$ and $x_2 \in d_B$, canonical representations imply $a = f_A(x_1) \in A_s$ and $b = f_B(x_2) \in B_s$. By the definition of operations in Minkowski space $(\circ)_m$:
			\[
			(\kgdt{A * B})_s = \{a * b \;:\; a \in A_s, \, b \in B_s, \, a * b \in \mathbb{K}\}
			\]
			Therefore, $y = a * b \in (\kgdt{A * B})_s$, which proves $f_{A*B}(d_A \times d_B) \subseteq (\kgdt{A * B})_s$.
			
			\item Backward Inclusion ($\supseteq$): 
			Let $z \in (\kgdt{A * B})_s$. By definition of Minkowski binary operations, there exist elements $a \in A_s$ and $b \in B_s$ such that $z = a * b$.
			Since $(f_A, d_A)$ and $(f_B, d_B)$ are canonical forms of $A$ and $B$, generating functions $f_A$ and $f_B$ map surjectively from their respective index domains onto $A_s$ and $B_s$. Thus:
			\begin{itemize}
				\item $\exists x_1^* \in d_A \text{ such that } f_A(x_1^*) = a$
				\item $\exists x_2^* \in d_B \text{ such that } f_B(x_2^*) = b$
			\end{itemize}
			Evaluating $f_{A*B}$ at $(x_1^*, x_2^*) \in d_A \times d_B$ yields:
			\[
			f_{A*B}(x_1^*, x_2^*) = f_A(x_1^*) * f_B(x_2^*) = a * b = z
			\]
			Hence $z \in f_{A*B}(d_A \times d_B)$, proving $(\kgdt{A * B})_s \subseteq f_{A*B}(d_A \times d_B)$.
		\end{enumerate}
		
		Combining both inclusions confirms that $f_{A*B}(d_A \times d_B) = (\kgdt{A * B})_s$, which completes the proof.
	\end{proof}

	\begin{proposition}
		Let $A \sim (f_A,d_A), B \sim (f_B,d_B)$ in $\tsbd{K}$, consider $a \in A$, then $a \in B$ if and only if 
		\[
		\exists (x,y) \in d_A \times d_B : a=f_A(x)=f_B(y)
		\]
	\end{proposition}
	\begin{proof}
		By the definition of the canonical form of uncertain numbers, for $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$, the corresponding sets of scenario values are represented via surjective generating functions:
		\[
		A_s = f_A(d_A) = \{f_A(x) : x \in d_A\} \quad \text{and} \quad B_s = f_B(d_B) = \{f_B(y) : y \in d_B\}
		\]
		
		We prove both directions of the logical equivalence:
		
		\begin{enumerate}[leftmargin=*]
			\item Forward Direction ($\Rightarrow$): 
			Assume $a \in A$ (meaning $a \in A_s$) and $a \in B$ (meaning $a \in B_s$).
			\begin{itemize}
				\item Since $a \in A_s = f_A(d_A)$, there exists at least one index $x \in d_A$ such that $a = f_A(x)$.
				\item Since $a \in B_s = f_B(d_B)$, there exists at least one index $y \in d_B$ such that $a = f_B(y)$.
			\end{itemize}
			Combining these two facts, for the scenario pair $(x, y) \in d_A \times d_B$, we obtain:
			\[
			a = f_A(x) = f_B(y)
			\]
			
			\item Backward Direction ($\Leftarrow$): 
			Assume there exists a pair $(x, y) \in d_A \times d_B$ such that $a = f_A(x) = f_B(y)$.
			\begin{itemize}
				\item Since $x \in d_A$, we have $f_A(x) \in f_A(d_A) = A_s \implies a \in A$.
				\item Since $y \in d_B$, we have $f_B(y) \in f_B(d_B) = B_s \implies a \in B$.
			\end{itemize}
			Thus, $a \in A$ and $a \in B$.
		\end{enumerate}
		
		This completes the proof.
	\end{proof}
	\begin{remark}
		This proposition transforms the essence of element checking $a \in B$ (or subset inclusion $A \subseteq B$) from cumbersome data lookups into finding roots of algebraic equation $f_B(y) = f_A(x)$ over index domain $d_A \times d_B$. This completely eliminates dependence on data size when sets expand.
	\end{remark}
	
	\begin{proposition}[Canonical Form of Intersection]
		\label{prop:dang_chuan_tac_phep_giao}
		Let $A\sim (f_A,d_A), B \sim (f_B,d_B)$ in $\tsbd{K}$. Then 
		\[
		C = A \cap B
		\]
		is an uncertain number with canonical form $(f_{A\cap B},d_{A\cap B})$ given by:
		\begin{align*}
			d_{A\cap B} &= \set{(x,y) \in d_A \times d_B : f_A(x) = f_B(y)} \\
			f_{A\cap B}(z) &= f_A(x) = f_B(y), z=(x,y) \in d_{A\cap B}
		\end{align*}
	\end{proposition}
	\begin{proof}
		By Proposition 2.1, an element $c \in \mathbb{K}$ belongs to the intersection $A \cap B$ if and only if there exists a scenario pair $(x, y) \in d_A \times d_B$ such that:
		\[
		c = f_A(x) = f_B(y)
		\]
		We verify that $(f_{A \cap B}, d_{A \cap B})$ satisfies the two requirements of a canonical form for $C = A \cap B$:
		
		\begin{enumerate}[leftmargin=*]
			\item Well-definedness of Index Domain and Generating Function: \\
			The index domain $d_{A \cap B} = \{(x, y) \in d_A \times d_B : f_A(x) = f_B(y)\}$ collects precisely all scenario pairs $(x, y)$ that produce common values between $A$ and $B$.
			
			The generating function $f_{A \cap B} : d_{A \cap B} \to \mathbb{K}$ defined by $f_{A \cap B}(z) = f_A(x) = f_B(y)$ for $z = (x, y) \in d_{A \cap B}$ is unambiguously well-defined because $f_A(x) = f_B(y)$ holds for every element in $d_{A \cap B}$.
			
			\item Set-Image Preservation: \\
			We establish $f_{A \cap B}(d_{A \cap B}) = (A \cap B)_s$ via mutual inclusion:
			\begin{itemize}[leftmargin=*]
				\item Forward Inclusion ($\subseteq$): 
				Let $c \in f_{A \cap B}(d_{A \cap B})$. By definition of $f_{A \cap B}$, there exists $z = (x, y) \in d_{A \cap B}$ such that $c = f_A(x) = f_B(y)$. 
				Since $x \in d_A$, $c = f_A(x) \in A_s$. Since $y \in d_B$, $c = f_B(y) \in B_s$. 
				Thus, $c \in A_s \cap B_s = (A \cap B)_s$, proving $f_{A \cap B}(d_{A \cap B}) \subseteq (A \cap B)_s$.
				
				\item Backward Inclusion ($\supseteq$): 
				Let $c \in (A \cap B)_s = A_s \cap B_s$. 
				Since $c \in A_s$, there exists $x^* \in d_A$ such that $f_A(x^*) = c$. 
				Since $c \in B_s$, there exists $y^* \in d_B$ such that $f_B(y^*) = c$. 
				Therefore, $f_A(x^*) = f_B(y^*)$, which implies $z^* = (x^*, y^*) \in d_{A \cap B}$. 
				Evaluating $f_{A \cap B}$ at $z^*$ yields $f_{A \cap B}(z^*) = f_A(x^*) = c \in f_{A \cap B}(d_{A \cap B})$. 
				This proves $(A \cap B)_s \subseteq f_{A \cap B}(d_{A \cap B})$.
			\end{itemize}
		\end{enumerate}
		
		Combining both inclusions confirms $f_{A \cap B}(d_{A \cap B}) = (A \cap B)_s$, which completes the proof.
	\end{proof}
	\begin{example}
		Consider two uncertain numbers $A, B \in \mathbf{U}(\mathbb{R})$ with canonical forms given by:
		\begin{itemize}[leftmargin=*]
			\item $A \sim (f_A, d_A)$ with $f_A(x) = 2x + 1$ on $d_A = \{1, 2, 3, 4\} \Rightarrow A = \{3, 5, 7, 9\}_u$.
			\item $B \sim (f_B, d_B)$ with $f_B(y) = 3y - 1$ on $d_B = \{1, 2, 3, 4\} \Rightarrow B = \{2, 5, 8, 11\}_u$.
		\end{itemize}
		
		According to Proposition \ref{prop:dang_chuan_tac_phep_giao}, we find index domain $d_{A \cap B}$ by solving algebraic equation on $d_A \times d_B$:
		\[
		f_A(x) = f_B(y) \iff 2x + 1 = 3y - 1 \iff 2x - 3y = -2
		\]
		Testing pairs $(x, y) \in \{1, 2, 3, 4\} \times \{1, 2, 3, 4\}$, the equation has unique solution $x = 2, y = 2$. Therefore:
		\[
		d_{A \cap B} = \{(2, 2)\}
		\]
		The corresponding generating function yields value:
		\[
		f_{A \cap B}(2, 2) = f_A(2) = 2(2) + 1 = 5
		\]
		Thus canonical form $(f_{A \cap B}, d_{A \cap B})$ accurately identifies $A \cap B = \{5\}_u$.
	\end{example}
	
	\begin{proposition}[Canonical Form of Union]
		Let $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$ in $\mathbf{U}(\mathbb{K})$. Canonical form $(f_{A \cup B}, d_{A \cup B})$ of uncertain number $A \cup B$ is defined by:
		\[
		\begin{cases}
			d_{A \cup B} = (d_A \times \{0\}) \cup (d_B \times \{1\}) \\[1ex]
			f_{A \cup B}(z) = \begin{cases} 
				f_A(x), & \text{if } z = (x, 0) \in d_A \times \{0\} \\[1ex]
				f_B(y), & \text{if } z = (y, 1) \in d_B \times \{1\}
			\end{cases}
		\end{cases}
		\]
	\end{proposition}
	\begin{proof}
		Let $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$ in $\mathbf{U}(\mathbb{K})$. To establish that $(f_{A \cup B}, d_{A \cup B})$ is a valid canonical form of $C = A \cup B$, we must verify that the index domain and the piecewise generating function preserve the set-image $(A \cup B)_s$.
		
		\begin{enumerate}[leftmargin=*]
			\item Well-definedness of Disjoint Index Domain: \\
			By defining the index domain as the disjoint union of indexed products:
			\[
			d_{A \cup B} = (d_A \times \{0\}) \cup (d_B \times \{1\})
			\]
			we guarantee that $(d_A \times \{0\}) \cap (d_B \times \{1\}) = \emptyset$, eliminating any index collision between domains $d_A$ and $d_B$. Thus, the piecewise definition of $f_{A \cup B}$ is unambiguous and well-defined for every $z \in d_{A \cup B}$.
			
			\item Set-Image Preservation: \\
			We show that $f_{A \cup B}(d_{A \cup B}) = (A \cup B)_s$ via mutual inclusion:
			\begin{itemize}[leftmargin=*]
				\item Forward Inclusion ($\subseteq$): 
				Let $c \in f_{A \cup B}(d_{A \cup B})$. Then there exists $z \in d_{A \cup B}$ such that $c = f_{A \cup B}(z)$. 
				\begin{itemize}
					\item If $z = (x, 0) \in d_A \times \{0\}$, then $c = f_A(x) \in f_A(d_A) = A_s \subseteq A_s \cup B_s$.
					\item If $z = (y, 1) \in d_B \times \{1\}$, then $c = f_B(y) \in f_B(d_B) = B_s \subseteq A_s \cup B_s$.
				\end{itemize}
				In either case, $c \in A_s \cup B_s = (A \cup B)_s$, proving $f_{A \cup B}(d_{A \cup B}) \subseteq (A \cup B)_s$.
				
				\item Backward Inclusion ($\supseteq$): 
				Let $c \in (A \cup B)_s = A_s \cup B_s$. 
				\begin{itemize}
					\item If $c \in A_s$, by surjectivity of $f_A$, there exists $x^* \in d_A$ such that $f_A(x^*) = c$. Choosing $z^* = (x^*, 0) \in d_{A \cup B}$, we have $f_{A \cup B}(z^*) = f_A(x^*) = c$.
					\item If $c \in B_s$, by surjectivity of $f_B$, there exists $y^* \in d_B$ such that $f_B(y^*) = c$. Choosing $z^* = (y^*, 1) \in d_{A \cup B}$, we have $f_{A \cup B}(z^*) = f_B(y^*) = c$.
				\end{itemize}
				Thus, $c \in f_{A \cup B}(d_{A \cup B})$ in both scenarios, proving $(A \cup B)_s \subseteq f_{A \cup B}(d_{A \cup B})$.
			\end{itemize}
		\end{enumerate}
		
		Combining both inclusions confirms $f_{A \cup B}(d_{A \cup B}) = (A \cup B)_s$, completing the proof.
	\end{proof}
	\begin{theorem}[Undefinedness]
		Let $*$ be any binary operation on $\tsbd{K}$, $A \in \tsbd{K}$. Then we have 
		\[A * \emptyset = \emptyset * A= \emptyset\]
		That is, any uncertain arithmetic binary operation involving the null number yields the null number (undefined).
	\end{theorem}
	\begin{proof}
		Let $*$ be an arbitrary binary operation on $\tsbd{K}$ and $A \in \tsbd{K}$. By definition of the null number, $\emptyset$ is the undefined uncertain number whose underlying set of scenario values is the empty set, i.e., $(\emptyset)_s = \emptyset$.
		
		We prove $A * \emptyset = \emptyset * A = \emptyset$ across both primary arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By the definition of binary operations in Minkowski space:
			\[
			((A * \emptyset)_m)_s = \{a * b \;:\; a \in A_s, \, b \in (\emptyset)_s, \, a * b \in \mathbb{K}\}
			\]
			Since $(\emptyset)_s = \emptyset$, there exists no element $b \in (\emptyset)_s$. Consequently, the Cartesian product $A_s \times (\emptyset)_s = \emptyset$, which implies that no valid pairs $(a, b)$ can be formed to evaluate $a * b$. Thus:
			\[
			((A * \emptyset)_m)_s = \emptyset \implies (A * \emptyset)_m = \emptyset
			\]
			By symmetric argument, $((\emptyset * A)_m)_s = \{b * a \;:\; b \in \emptyset, \, a \in A_s\} = \emptyset \implies (\emptyset * A)_m = \emptyset$.
			
			\item In Single-Point Arithmetic Space $(\circ)_1$: \\
			Consider an uncertain function $f(X, Y) = \kgdd{X * Y}$. The evaluation pass requires selecting representative points $x \in A_s$ and $y \in (\emptyset)_s$. Since $(\emptyset)_s = \emptyset$, no representative point $y$ can be assigned to the variable $\emptyset$. Hence, the pointwise mapping pass cannot be executed, yielding an empty image set:
			\[
			\kgdd{A * \emptyset} = \s{x * y \;:\; x \in A_s, \, y \in \emptyset} = \emptyset
			\]
		\end{enumerate}
		
		In all cases, any arithmetic binary operation involving the null number $\emptyset$ results in the null number $\emptyset$. This completes the proof.
	\end{proof}

	\begin{md}[Distributive Property of Operation $*$ with Union Operation]
		Let $*$ be a binary operation on uncertain numbers, $A$ and $B$ be uncertain numbers, we have
		\[A*B = \bigcup_{b \in B} {A * b} \]
	\end{md}
	\begin{proof}
		Let $*$ be a binary operation on $\tsbd{K}$, and let $A, B \in \tsbd{K}$ be uncertain numbers with scenario value sets $A_s$ and $B_s$, respectively. 
		
		We prove the identity $A * B = \bigcup_{b \in B} (A * b)$ by demonstrating mutual inclusion of their underlying scenario value sets across arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By definition of binary operations in Minkowski space, the set of scenario values of $(A * B)_m$ is given by:
			\[
			((A * B)_m)_s = \{a * b \;:\; a \in A_s, \, b \in B_s\}
			\]
			On the other hand, for a fixed element $b \in B_s$, identifying $b$ as a single-valued uncertain number $\{b\}_u$, the term $A * b$ has scenario value set:
			\[
			(A * b)_s = \{a * b \;:\; a \in A_s\}
			\]
			Taking the set union over all $b \in B_s$:
			\[
			\left( \bigcup_{b \in B} (A * b) \right)_s = \bigcup_{b \in B_s} (A * b)_s = \bigcup_{b \in B_s} \{a * b \;:\; a \in A_s\} = \{a * b \;:\; a \in A_s, \, b \in B_s\}
			\]
			Thus, $((A * B)_m)_s = \left( \bigcup_{b \in B} (A * b) \right)_s$.
			
			\item In Single-Point Arithmetic Space $(\circ)_1$: \\
			Let $(f_A, d_A)$ and $(f_B, d_B)$ be canonical forms of $A$ and $B$. Under the single-point evaluation operator $\kgdd{\circ}$, during each evaluation pass, variable $B$ takes a representative element $b = f_B(y) \in B_s$ for $y \in d_B$. The resulting image set decomposes along the fiber of elements in $B$:
			\[
			[\kgdd{A * B}]_s = \{f_A(x) * f_B(y) \;:\; x \in d_A, \, y \in d_B\} = \bigcup_{y \in d_B} \{f_A(x) * f_B(y) \;:\; x \in d_A\}
			\]
			Substituting $b = f_B(y) \in B_s$, this equals $\bigcup_{b \in B_s} (A * b)_s$.
		\end{enumerate}
		
		In all arithmetic spaces, the set of scenario values matches identically, confirming that $A * B = \bigcup_{b \in B} (A * b)$.
	\end{proof}
	\begin{theorem}
		Let $A \in \tsbd{K}$, we set 
		\begin{enumerate}[leftmargin=*]
			\item $Id_+=\set{X-X: X \in \tsbd{K}}$
			\item $Id_*=\set{X/X: X \in \tsbd{K}}$
		\end{enumerate}
		From which we have: 
		\begin{align*}
			(\forall e_+ \in Id_+)(A + e_+ &= A - e_+) \\
			(\forall e_* \in Id_*)(A \cdot e_* &= A / e_*)
		\end{align*}
	\end{theorem}
	\begin{proof}
		Let $A \in \tsbd{K}$. By definition, any elements $e_+ \in Id_+$ and $e_* \in Id_*$ are constructed from some uncertain numbers $X, Y \in \tsbd{K}$ as $e_+ = X - X$ and $e_* = Y / Y$, respectively.
		
		We prove the two identities separately across the relevant arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item Proof of $A + e_+ = A - e_+$ for all $e_+ \in Id_+$: \\
			Let $e_+ = X - X \in Id_+$ for some $X \in \tsbd{K}$. The scenario value set of $e_+$ is given by:
			\[
			(e_+)_s = \{x_1 - x_2 \;:\; x_1, x_2 \in X_s\}
			\]
			Notice that if $v \in (e_+)_s$, then $v = x_1 - x_2$ for some $x_1, x_2 \in X_s$. Reversing the order gives $-v = x_2 - x_1 \in (e_+)_s$. Thus, $(e_+)_s$ is a symmetric set around zero, meaning $(e_+)_s = -(e_+)_s$.
			
			Evaluating $A + e_+$ and $A - e_+$ in Minkowski arithmetic space $(\circ)_m$:
			\begin{align*}
				(A + e_+)_s &= \{a + v \;:\; a \in A_s, \, v \in (e_+)_s\} \\
				(A - e_+)_s &= \{a - v \;:\; a \in A_s, \, v \in (e_+)_s\}
			\end{align*}
			Since $(e_+)_s = -(e_+)_s$, as $v$ ranges over $(e_+)_s$, $-v$ also ranges over the exact same set $(e_+)_s$. Therefore:
			\[
			(A - e_+)_s = \{a - (-u) \;:\; a \in A_s, \, u \in (e_+)_s\} = \{a + u \;:\; a \in A_s, \, u \in (e_+)_s\} = (A + e_+)_s
			\]
			Hence, $A + e_+ = A - e_+$ holds.
			
			\item Proof of $A \cdot e_* = A / e_*$ for all $e_* \in Id_*$: \\
			Let $e_* = Y / Y \in Id_*$ for some $Y \in \tsbd{K}$ where $0 \notin Y_s$. The scenario value set of $e_*$ is:
			\[
			(e_*)_s = \left\{ \frac{y_1}{y_2} \;:\; y_1, y_2 \in Y_s \right\}
			\]
			For any element $w \in (e_*)_s$, $w = \frac{y_1}{y_2}$. Taking its multiplicative inverse yields $\frac{1}{w} = \frac{y_2}{y_1} \in (e_*)_s$. Thus, $(e_*)_s$ is closed under inversion, meaning $(e_*)_s = (e_*)_s^{-1}$.
			
			Evaluating $A \cdot e_*$ and $A / e_*$ in Minkowski arithmetic space $(\circ)_m$:
			\begin{align*}
				(A \cdot e_*)_s &= \{a \cdot w \;:\; a \in A_s, \, w \in (e_*)_s\} \\
				(A / e_*)_s &= \left\{ \frac{a}{w} \;:\; a \in A_s, \, w \in (e_*)_s \right\}
			\end{align*}
			Since $(e_*)_s = (e_*)_s^{-1}$, replacing $w$ with $\frac{1}{u}$ (where $u \in (e_*)_s$) yields:
			\[
			(A / e_*)_s = \left\{ \frac{a}{1/u} \;:\; a \in A_s, \, u \in (e_*)_s \right\} = \{a \cdot u \;:\; a \in A_s, \, u \in (e_*)_s\} = (A \cdot e_*)_s
			\]
			Hence, $A \cdot e_* = A / e_*$ holds.
		\end{enumerate}				
		In both cases, the structural symmetry of the identity-like sets $Id_+$ under negation and $Id_*$ under inversion ensures the equivalence of the operations. This completes the proof.
	\end{proof}
	\begin{theorem}[Inverse]
		Consider $A \in \tsbd{K}, \dimU(A) > 1$, then:
		\begin{enumerate}[leftmargin=*]
			\item There exists no uncertain number B such that $A+B = 0$
			\item There exists no uncertain number B such that $AB = 1$
		\end{enumerate}
	\end{theorem}
	\begin{proof}
		Let $A \in \tsbd{K}$ be an uncertain number with scenario value set $A_s$ and dimension $\dimU(A) > 1$.
		
		We prove the non-existence of additive and multiplicative inverses by contradiction:
		
		\begin{enumerate}[leftmargin=*]
			\item Proof of Part 1 (Non-existence of Additive Inverse): \\
			Suppose there exists an uncertain number $B \in \tsbd{K}$ with scenario value set $B_s$ such that $A + B = 0$.
			
			By definition of addition in Minkowski arithmetic space $(\circ)_m$:
			\[
			(A + B)_s = \{a + b \;:\; a \in A_s, \, b \in B_s\} = \{0\}
			\]
			This implies that for every pair $(a, b) \in A_s \times B_s$, we must have $a + b = 0$, or equivalently $b = -a$.
			
			Since $\dimU(A) > 1$, there exist at least two distinct elements $a_1, a_2 \in A_s$ with $a_1 \neq a_2$. 
			For any element $b \in B_s$, we must simultaneously satisfy:
			\[
			b = -a_1 \quad \text{and} \quad b = -a_2
			\]
			which implies $-a_1 = -a_2 \iff a_1 = a_2$, contradicting $a_1 \neq a_2$. Thus, no such uncertain number $B$ exists.
			
			\item Proof of Part 2 (Non-existence of Multiplicative Inverse): \\
			Suppose there exists an uncertain number $B \in \tsbd{K}$ with scenario value set $B_s$ such that $AB = 1$.
			
			By definition of multiplication in Minkowski arithmetic space $(\circ)_m$:
			\[
			(AB)_s = \{a \cdot b \;:\; a \in A_s, \, b \in B_s\} = \{1\}
			\]
			This implies that $0 \notin A_s$ (otherwise $0 \cdot b = 0 \neq 1$) and for every pair $(a, b) \in A_s \times B_s$, we must have $a \cdot b = 1$, or equivalently $b = \frac{1}{a}$.
			
			Since $\dimU(A) > 1$, there exist at least two distinct non-zero elements $a_1, a_2 \in A_s$ with $a_1 \neq a_2$.
			For any element $b \in B_s$, we must simultaneously satisfy:
			\[
			b = \frac{1}{a_1} \quad \text{and} \quad b = \frac{1}{a_2}
			\]
			which implies $\frac{1}{a_1} = \frac{1}{a_2} \iff a_1 = a_2$, contradicting $a_1 \neq a_2$. Thus, no such uncertain number $B$ exists.
		\end{enumerate}
		
		In conclusion, any uncertain number $A$ with $\dimU(A) > 1$ lacks both additive and multiplicative inverses in $\tsbd{K}$.
	\end{proof}
	\begin{theorem}
		Let $A$ and $B$ be two uncertain numbers, we have 
		$$\rad(A+B) \leq \rad(A) +\rad(B)$$
	\end{theorem}
	\begin{proof}
		Let $A, B \in \tsbd{K}$ be two uncertain numbers. By Definition 2.6, the uncertain radius of any uncertain number $X \in \tsbd{K}$ is defined by:
		\[
		\rad(X) = \frac{1}{2} \sup \set{x_1 - x_2 \;:\; x_1, x_2 \in X_s}
		\]
		
		By definition of addition in Minkowski arithmetic space $(\circ)_m$, the scenario value set of $A + B$ is:
		\[
		(A + B)_s = \set{a + b \;:\; a \in A_s, \, b \in B_s}
		\]
		Let $z_1, z_2 \in (A + B)_s$ be any two arbitrary elements. Then there exist $a_1, a_2 \in A_s$ and $b_1, b_2 \in B_s$ such that $z_1 = a_1 + b_1$ and $z_2 = a_2 + b_2$.
		
		Evaluating the difference $z_1 - z_2$:
		\[
		z_1 - z_2 = (a_1 + b_1) - (a_2 + b_2) = (a_1 - a_2) + (b_1 - b_2)
		\]
		
		Taking the supremum over all possible pairs $(z_1, z_2) \in (A + B)_s \times (A + B)_s$:
		\begin{align*}
			\sup_{z_1, z_2 \in (A + B)_s} (z_1 - z_2) &= \sup_{\substack{a_1, a_2 \in A_s \\ b_1, b_2 \in B_s}} \left[ (a_1 - a_2) + (b_1 - b_2) \right] \\
			&\le \sup_{a_1, a_2 \in A_s} (a_1 - a_2) + \sup_{b_1, b_2 \in B_s} (b_1 - b_2)
		\end{align*}
		
		Multiplying both sides of the inequality by $\frac{1}{2}$:
		\[
		\frac{1}{2} \sup_{z_1, z_2 \in (A + B)_s} (z_1 - z_2) \le \frac{1}{2} \sup_{a_1, a_2 \in A_s} (a_1 - a_2) + \frac{1}{2} \sup_{b_1, b_2 \in B_s} (b_1 - b_2)
		\]
		
		Applying the definition of uncertain radius to each term yields:
		\[
		\rad(A + B) \le \rad(A) + \rad(B)
		\]
		This completes the proof.
	\end{proof}
	\begin{theorem}
		Let $A$ be an uncertain number, $b$ be a single-valued constant. We have 
		$$\rad(A+b) = \rad(A)$$
		This theorem allows an object with uncertain position to preserve its uncertain radius under rigid translation.
	\end{theorem}
	\begin{proof}
		Let $A \in \tsbd{K}$ be an uncertain number and $b \in \mathbb{K}$ be a single-valued constant (identified with the single-point uncertain number $\{b\}_u$).
		
		By Definition 2.6, the scenario value set of $A + b$ in Minkowski arithmetic space $(\circ)_m$ is given by:
		\[
		(A + b)_s = \{a + b \;:\; a \in A_s\}
		\]
		
		Applying the definition of uncertain radius to $A + b$:
		\[
		\rad(A + b) = \frac{1}{2} \sup \set{z_1 - z_2 \;:\; z_1, z_2 \in (A + b)_s}
		\]
		
		Since any elements $z_1, z_2 \in (A + b)_s$ can be represented as $z_1 = a_1 + b$ and $z_2 = a_2 + b$ for some $a_1, a_2 \in A_s$, we evaluate their difference:
		\[
		z_1 - z_2 = (a_1 + b) - (a_2 + b) = a_1 - a_2
		\]
		
		Notice that the constant term $b$ cancels out completely. Taking the supremum over all pairs $(a_1, a_2) \in A_s \times A_s$:
		\[
		\sup_{z_1, z_2 \in (A + b)_s} (z_1 - z_2) = \sup_{a_1, a_2 \in A_s} (a_1 - a_2)
		\]
		
		Multiplying both sides by $\frac{1}{2}$:
		\[
		\rad(A + b) = \frac{1}{2} \sup_{a_1, a_2 \in A_s} (a_1 - a_2) = \rad(A)
		\]
		
		Hence, $\rad(A + b) = \rad(A)$, which proves that rigid translation preserves the uncertain radius of an uncertain object.
	\end{proof}
	\begin{theorem}[Realistic Similarity]
		Let $A$ be an uncertain number and $k$ be a single-valued constant, we have: $\rad(k A) = |k| \rad(A)$
	\end{theorem}
	\begin{proof}
		Let $A \in \tsbd{K}$ be an uncertain number with scenario value set $A_s$, and $k \in \mathbb{K}$ be a single-valued constant (identified with $\{k\}_u$).
		
		By definition of scalar multiplication in Minkowski arithmetic space $(\circ)_m$, the scenario value set of $k A$ is given by:
		\[
		(k A)_s = \{k \cdot a \;:\; a \in A_s\}
		\]
		
		Applying the definition of uncertain radius (Definition 2.6) to $k A$:
		\[
		\rad(k A) = \frac{1}{2} \sup \set{z_1 - z_2 \;:\; z_1, z_2 \in (k A)_s}
		\]
		
		Since any elements $z_1, z_2 \in (k A)_s$ can be written as $z_1 = k a_1$ and $z_2 = k a_2$ for some $a_1, a_2 \in A_s$, we evaluate their difference:
		\[
		z_1 - z_2 = k a_1 - k a_2 = k(a_1 - a_2)
		\]
		
		We consider two cases based on the sign of $k \in \mathbb{R}$:
		\begin{enumerate}[leftmargin=*]
			\item Case 1: $k \ge 0$. \\
			Since $k$ is non-negative, taking the supremum over all pairs $a_1, a_2 \in A_s$ preserves the order:
			\[
			\sup_{a_1, a_2 \in A_s} [k(a_1 - a_2)] = k \sup_{a_1, a_2 \in A_s} (a_1 - a_2)
			\]
			Thus:
			\[
			\rad(k A) = \frac{1}{2} \cdot k \sup_{a_1, a_2 \in A_s} (a_1 - a_2) = k \cdot \rad(A) = |k| \rad(A)
			\]
			
			\item Case 2: $k < 0$. \\
			Rewriting $k = -|k|$, the difference becomes $z_1 - z_2 = -|k|(a_1 - a_2) = |k|(a_2 - a_1)$. \\
			As $a_1, a_2$ range independently over $A_s$, $a_2 - a_1$ ranges over the exact same set of differences as $a_1 - a_2$. Therefore:
			\[
			\sup_{a_1, a_2 \in A_s} [-|k|(a_1 - a_2)] = |k| \sup_{a_2, a_1 \in A_s} (a_2 - a_1) = |k| \sup_{a_1, a_2 \in A_s} (a_1 - a_2)
			\]
			Thus:
			\[
			\rad(k A) = \frac{1}{2} \cdot |k| \sup_{a_1, a_2 \in A_s} (a_1 - a_2) = |k| \rad(A)
			\]
		\end{enumerate}
		
		Combining both cases confirms that $\rad(k A) = |k| \rad(A)$ for any single-valued constant $k$. This completes the proof.
	\end{proof}
	\begin{theorem}[Identity Theorem with Classical Arithmetic]
		Let $A = \s{a} , B = \s{b}$ and * be a binary operation on $\mathbb{R}$ where $a*b$ is defined, then 
		$$A * B = \s{a*b} = a*b$$
	\end{theorem}
	\begin{proof}
		Let $A = \s{a}$ and $B = \s{b}$ be two single-valued (deterministic) uncertain numbers, where $a, b \in \mathbb{R}$. Their underlying sets of scenario values are singleton sets:
		\[
		A_s = \{a\} \quad \text{and} \quad B_s = \{b\}
		\]
		
		We prove the identity across both Minkowski and Single-point arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By definition of binary operations in Minkowski space:
			\[
			(A * B)_s = \{x * y \;:\; x \in A_s, \, y \in B_s, \, x * y \in \mathbb{R}\}
			\]
			Since $A_s = \{a\}$ and $B_s = \{b\}$, the Cartesian product $A_s \times B_s = \{(a, b)\}$ contains exactly one pair. Assuming $a * b$ is well-defined in classical arithmetic:
			\[
			(A * B)_s = \{a * b\}
			\]
			Transforming back to an uncertain number using the transfer operator yields:
			\[
			A * B = (\{a * b\})_u = \s{a * b} \equiv a * b
			\]
			
			\item In Single-Point Arithmetic Space $(\circ)_1$: \\
			Let $(f_A, d_A)$ and $(f_B, d_B)$ be canonical forms of $A$ and $B$, where $d_A = d_B = \{1\}$ and $f_A(1) = a, f_B(1) = b$. 
			Evaluating the pointwise operation on singleton index domains:
			\[
			[\kgdd{A * B}]_s = \{f_A(1) * f_B(1)\} = \{a * b\} \implies \kgdd{A * B} = \s{a * b} \equiv a * b
			\]
		\end{enumerate}
		
		Thus, when restricted to single-valued operands, uncertain arithmetic reduces identically to classical arithmetic. This completes the proof.
	\end{proof}
	\begin{theorem}[Commutativity]
		Let $A,B$ be two uncertain numbers and * be a commutative binary operation on $\mathbb{R}$, we have
		$$A*B = B*A$$
	\end{theorem}
	\begin{proof}
		Let $A, B \in \tsbd{K}$ be two uncertain numbers with scenario value sets $A_s$ and $B_s$, respectively. Let $*$ be a commutative binary operation on $\mathbb{R}$ (i.e., $a * b = b * a$ for all $a, b \in \mathbb{R}$).
		
		We prove the identity $A * B = B * A$ across both primary arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By the definition of binary operations in Minkowski space:
			\[
			((A * B)_m)_s = \{a * b \;:\; a \in A_s, \, b \in B_s, \, a * b \in \mathbb{R}\}
			\]
			Since $*$ is commutative in $\mathbb{R}$, we have $a * b = b * a$ for every pair $(a, b) \in A_s \times B_s$. Thus:
			\[
			((A * B)_m)_s = \{b * a \;:\; b \in B_s, \, a \in A_s, \, b * a \in \mathbb{R}\} = ((B * A)_m)_s
			\]
			Applying the transfer operator yields $(A * B)_m = (B * A)_m$.
			
			\item In Single-Point Arithmetic Space $(\circ)_1$: \\
			Let $(f_A, d_A)$ and $(f_B, d_B)$ be canonical forms of $A$ and $B$, respectively. Under the single-point evaluation operator $\kgdd{\circ}$, during each evaluation pass at index pair $(x, y) \in d_A \times d_B$:
			\[
			[\kgdd{A * B}]_s = \{f_A(x) * f_B(y) \;:\; x \in d_A, \, y \in d_B\}
			\]
			By commutativity of $*$ on individual values, $f_A(x) * f_B(y) = f_B(y) * f_A(x)$. Reordering the elements yields:
			\[
			[\kgdd{A * B}]_s = \{f_B(y) * f_A(x) \;:\; y \in d_B, \, x \in d_A\} = [\kgdd{B * A}]_s
			\]
			Hence, $\kgdd{A * B} = \kgdd{B * A}$.
		\end{enumerate}
		
		In all arithmetic spaces, the sets of scenario values match identically, confirming that $A * B = B * A$.
	\end{proof}
	\begin{theorem}[Associativity]
		Let $A,B,C$ be uncertain numbers and * be an associative binary operation on $\mathbb{R}$, then:
		\[A*(B*C) = (A*B)*C\]
	\end{theorem}
	\begin{proof}
		Let $A, B, C \in \tsbd{K}$ be three uncertain numbers with scenario value sets $A_s, B_s$, and $C_s$, respectively. Let $*$ be an associative binary operation on $\mathbb{R}$ (i.e., $a * (b * c) = (a * b) * c$ for all $a, b, c \in \mathbb{R}$).
		
		We prove the identity $A * (B * C) = (A * B) * C$ across both primary arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By the definition of binary operations in Minkowski space, evaluating $A * (B * C)$ yields scenario value set:
			\begin{align*}
				((A * (B * C))_m)_s &= \{a * u \;:\; a \in A_s, \, u \in ((B * C)_m)_s\} \\
				&= \{a * (b * c) \;:\; a \in A_s, \, b \in B_s, \, c \in C_s\}
			\end{align*}
			Similarly, evaluating $(A * B) * C$ yields:
			\begin{align*}
				(( (A * B) * C)_m)_s &= \{v * c \;:\; v \in ((A * B)_m)_s, \, c \in C_s\} \\
				&= \{(a * b) * c \;:\; a \in A_s, \, b \in B_s, \, c \in C_s\}
			\end{align*}
			Since $*$ is associative in $\mathbb{R}$, $a * (b * c) = (a * b) * c$ holds for every tuple $(a, b, c) \in A_s \times B_s \times C_s$. Thus:
			\[
			((A * (B * C))_m)_s = (( (A * B) * C)_m)_s
			\]
			Applying the transfer operator yields $(A * (B * C))_m = ((A * B) * C)_m$.
			
			\item In Single-Point Arithmetic Space $(\circ)_1$: \\
			Let $(f_A, d_A), (f_B, d_B)$, and $(f_C, d_C)$ be canonical forms of $A, B$, and $C$, respectively. Under the single-point evaluation operator $\kgdd{\circ}$, at index tuple $(x, y, z) \in d_A \times d_B \times d_C$:
			\[
			[\kgdd{A * (B * C)}]_s = \{f_A(x) * (f_B(y) * f_C(z)) \;:\; x \in d_A, \, y \in d_B, \, z \in d_C\}
			\]
			By associativity of $*$ on individual real values:
			\[
			f_A(x) * (f_B(y) * f_C(z)) = (f_A(x) * f_B(y)) * f_C(z)
			\]
			Therefore:
			\[
			[\kgdd{A * (B * C)}]_s = \{(f_A(x) * f_B(y)) * f_C(z) \;:\; x \in d_A, \, y \in d_B, \, z \in d_C\} = [\kgdd{(A * B) * C}]_s
			\]
			Hence, $\kgdd{A * (B * C)} = \kgdd{(A * B) * C}$.
		\end{enumerate}
		
		In all arithmetic spaces, the sets of scenario values match identically, confirming $A * (B * C) = (A * B) * C$.
	\end{proof}
	\begin{theorem}[Distributivity]
		Let $K,A,B$ be uncertain numbers, we have:
		\[K(A+B)\ = \bigcup_{k \in K} [k(A+B)] = \bigcup_{k \in K} (kA+kB)\]
		\[(AB)^K\ = \bigcup_{k \in K} \left[(AB)^k \right] = \bigcup_{k \in K} \left(A^kB^k\right) \]
	\end{theorem}
	\begin{proof}
		Let $K, A, B \in \tsbd{K}$ be uncertain numbers with scenario value sets $K_s, A_s$, and $B_s$, respectively. 
		
		We prove both distributive identities separately across the relevant arithmetic spaces:
		
		\begin{enumerate}[leftmargin=*]
			\item Proof of $K(A + B) = \bigcup_{k \in K} [k(A + B)] = \bigcup_{k \in K} (kA + kB)$: \\
			By definition of arithmetic operations in Minkowski space $(\circ)_m$, evaluating $K(A + B)$ yields the set of scenario values:
			\[
			(K(A + B))_s = \{k \cdot u \;:\; k \in K_s, \, u \in (A + B)_s\} = \{k(a + b) \;:\; k \in K_s, \, a \in A_s, \, b \in B_s\}
			\]
			Applying the distributive law of real numbers $k(a + b) = ka + kb$ for each element tuple $(k, a, b) \in K_s \times A_s \times B_s$:
			\begin{align*}
				(K(A + B))_s &= \{ka + kb \;:\; k \in K_s, \, a \in A_s, \, b \in B_s\} \\
				&= \bigcup_{k \in K_s} \{ka + kb \;:\; a \in A_s, \, b \in B_s\} \\
				&= \bigcup_{k \in K_s} (k A + k B)_s
			\end{align*}
			Transforming back via the transfer operator gives:
			\[
			K(A + B) = \bigcup_{k \in K} [k(A + B)] = \bigcup_{k \in K} (kA + kB)
			\]
			
			\item Proof of $(AB)^K = \bigcup_{k \in K} \left[(AB)^k\right] = \bigcup_{k \in K} \left(A^k B^k\right)$: \\
			By definition of exponentiation and multiplication in Minkowski space $(\circ)_m$, evaluating $(AB)^K$ yields the set of scenario values:
			\[
			((AB)^K)_s = \{v^k \;:\; v \in (AB)_s, \, k \in K_s\} = \{(a \cdot b)^k \;:\; a \in A_s, \, b \in B_s, \, k \in K_s\}
			\]
			Applying the power law of real numbers $(a \cdot b)^k = a^k \cdot b^k$ (for $a, b > 0$):
			\begin{align*}
				((AB)^K)_s &= \{a^k \cdot b^k \;:\; a \in A_s, \, b \in B_s, \, k \in K_s\} \\
				&= \bigcup_{k \in K_s} \{a^k \cdot b^k \;:\; a \in A_s, \, b \in B_s\} \\
				&= \bigcup_{k \in K_s} (A^k B^k)_s
			\end{align*}
			Transforming back via the transfer operator gives:
			\[
			(AB)^K = \bigcup_{k \in K} \left[(AB)^k\right] = \bigcup_{k \in K} \left(A^k B^k\right)
			\]
		\end{enumerate}
		
		In all cases, the underlying set-theoretic images match identically, confirming both distributive properties.
	\end{proof}

	\section{Weighted Uncertain Numbers}
	
	\begin{definition}[Weight]
		Let $A$ be a set, a function $\ts{A}$ from $A$ to $[0,1]$ satisfying the following properties is called a weight of A: 
		\begin{enumerate}[leftmargin=*]
			\item $$\sum_{a \in A}{\ts{A}(a) } = 1$$
			\item For all $a \in A$, $\ts{A}(a) > 0 $
		\end{enumerate}
	\end{definition}
	
	\begin{definition}[Weighted Uncertain Number over Field $\mathbb{K}$]
		A pair $(A,\ts{A})$ with $A \in \tsbd{K}$ and $\ts{A}$ being a weight of $A$ is called a weighted uncertain number over field $\mathbb{K}$. The set of all weighted uncertain numbers over field $\mathbb{K}$ is denoted by $\tsbdts{K}$.
	\end{definition}
	
	\begin{example}
		Let $A \in \tsbd{R}$ and $A = \s{a} = a $ with arbitrary weight $\ts{A}$, we always have $A = a$ with $\ts{A}(a) =1$. Such a number $A$ is a real number.
	\end{example}
	
	\begin{example}
		Set $A=\s{2k : k \in \mathbb{N}}$ together with weight $\ts{A}(2k) = \frac{1}{2^{k+1}}$ is an uncertain number over real field with weight $\ts{A}$.
	\end{example}
	
	\begin{theorem}[Weight of an Operation]
		Let * be a binary operation on $\tsbd{K}$, we define function 
		$$\ts{A*B}(c) := \sum_{a*b=c}{\ts{A}(a)\ts{B}(b)}$$ 
		with $c = a * b \in A * B $, where $\ts{A}$ is a weight of $A$ and $\ts{B}$ is a weight of $B$. Then $\ts{A*B}$ is also a weight of $A *B $.
	\end{theorem}
	\begin{proof}
		Let $(A, \ts{A})$ and $(B, \ts{B})$ be two weighted uncertain numbers in $\tsbdts{K}$, and let $*$ be a binary operation on $\tsbd{K}$. 
		
		By Definition 2.11, to prove that $\ts{A*B}$ is a valid weight of the resulting uncertain number $C = A * B$, we must verify two defining conditions: positivity for all elements in $C$ and total summation equality to $1$.
		
		\begin{enumerate}[leftmargin=*]
			\item Condition 1: Positivity ($\ts{A*B}(c) > 0$ for all $c \in A * B$). \\
			For any element $c \in A * B$, by definition of Minkowski binary operations, there exists at least one pair $(a, b) \in A \times B$ such that $a * b = c$. 
			
			Since $\ts{A}$ is a weight of $A$ and $\ts{B}$ is a weight of $B$, by Definition 2.11 we have $\ts{A}(a) > 0$ for all $a \in A$ and $\ts{B}(b) > 0$ for all $b \in B$. Consequently, for any pair $(a, b)$ satisfying $a * b = c$, the product of their weights is strictly positive:
			\[
			\ts{A}(a) \ts{B}(b) > 0
			\]
			Since the formula $\ts{A*B}(c) = \sum_{a*b=c} \ts{A}(a)\ts{B}(b)$ is a sum of strictly positive terms over a non-empty set of pairs, it follows that:
			\[
			\ts{A*B}(c) > 0, \quad \forall c \in A * B
			\]
			
			\item Condition 2: Normalization / Summation to 1 ($\sum_{c \in A*B} \ts{A*B}(c) = 1$). \\
			Summing the weight function $\ts{A*B}(c)$ over all elements $c \in A * B$:
			\[
			\sum_{c \in A*B} \ts{A*B}(c) = \sum_{c \in A*B} \left( \sum_{a*b=c} \ts{A}(a)\ts{B}(b) \right)
			\]
			Since every pair $(a, b) \in A \times B$ maps to a unique element $c = a * b \in A * B$, partitioning the Cartesian product $A \times B$ according to the level sets of $a * b = c$ covers the entire set $A \times B$ without overlap or omission. Thus, we can rewrite the double summation as a single summation over $A \times B$:
			\[
			\sum_{c \in A*B} \left( \sum_{a*b=c} \ts{A}(a)\ts{B}(b) \right) = \sum_{(a, b) \in A \times B} \ts{A}(a)\ts{B}(b)
			\]
			By Fubini-Tonelli theorem for discrete product spaces, the joint summation factors into independent marginal summations:
			\[
			\sum_{(a, b) \in A \times B} \ts{A}(a)\ts{B}(b) = \left( \sum_{a \in A} \ts{A}(a) \right) \cdot \left( \sum_{b \in B} \ts{B}(b) \right)
			\]
			Since $\ts{A}$ and $\ts{B}$ are valid weights, $\sum_{a \in A} \ts{A}(a) = 1$ and $\sum_{b \in B} \ts{B}(b) = 1$. Therefore:
			\[
			\sum_{c \in A*B} \ts{A*B}(c) = 1 \cdot 1 = 1
			\]
		\end{enumerate}
		
		Both conditions are satisfied, confirming that $\ts{A*B}$ is indeed a well-defined weight on the uncertain number $A * B$.
	\end{proof}
	\begin{definition}[Weighted Weak Binary Relation over Field $\mathbb{K}$]
		Let $\mathcal{R}$ be a binary relation on field $\mathbb{K}$. $(A,\ts{A}),(B,\ts{B}) \in \tsbdts{K}$. We define the weighted weak binary relation over field $\mathbb{K}$ as follows:
		\[\ddclmd(A\mathcal{R}B) := \sum_{(a,b) \in A \times B}{\left[\ts{A}(a)\ts{B}(b)I(a,b)\right]}\]
		Where I(a,b) is given by 
		\[
		I(a,b) := \begin{cases}
			1 & \text{if } (a,b) \in \mathcal{R} \\
			0 & \text{if } (a,b) \notin \mathcal{R}
		\end{cases}
		\]
	\end{definition}

			\item Given $f(X)=\kgdd{X^2}$, we define: \[\kgdd{\sqrt{X}} = \s{A \in \tsbd{K} : \kgdd{X^2} = X}\]
			\item Given $f(X)=\kgdd{\sin{X}}$, we define: \[\kgdd{\arcsin{X}} = \s{A \in \tsbd{K} : \kgdd{\sin(A)} = X}\]
		\end{enumerate}
	\end{example}

	\section{Graphs and Structure of Uncertain Functions}
	
	\begin{definition}[Graph of an Uncertain Function at a Point]
		Let $f : \mathcal{X} \to \mathcal{Y}$ be an uncertain function, $A \in \mathcal{X}$, we define the graph of the uncertain function $f$ at point $A$ as:
		\[\pgr_f(A):= A \times f(A)\]
	\end{definition}
	
	\begin{definition}[Graph of an Uncertain Function]
		Let $\mathcal{X} ,\mathcal{Y}$ be subsets of $\tsbd{K}$ where $\mathbb{K}$ is an arbitrary field.
		Let $f : \mathcal{X} \to \mathcal{Y}$ be an uncertain function, $A \in \mathcal{X}$, we define the graph of the uncertain function $f$ as:
		\[\gr{f}:= \set{\pgr_f(X) : X \in \mathcal{X}}\]
	\end{definition}
	
	\begin{example}
		For $f = X+X$, we have \[\pgr_f{\s{1,2}} = \s{(1,2),(1,3),(1,4),(2,2),(2,3),(2,4)}\]
	\end{example}
	
	\begin{definition}[Trace]
		Let $f$ be an uncertain function from $\mathcal{X}$ into $\mathcal{Y}$ mapping $X \mapsto Y = f(X)$.
		
		Considering $y \in Y = f(X)$, we define the trace of $y$ in the expression of $f(X)$, denoted as $\Tr(y \in f(X))$, as the set of all formal expressions whose evaluated result equals $y$, obtained by computing $f(X)$.
	\end{definition}
	
	\section{Uncertain Equations and Solution Theory}
	
	\begin{definition}[Uncertain Equation]
		Let $f : \mathcal{X}\to \mathcal{Y}$ be an uncertain function, $C$ be an uncertain constant; an uncertain equation with unknown $X$ is an equation of the form: \[f(X) = C\]
	\end{definition}
	
	\begin{example}
		Given the uncertain equation \[\kgdd{X^2+5X+6} = 0 \]
		It is easy to see that $X = \s{-2,-3}$ is a solution to this equation. 
		
		However, $X = \s{-2,-3}$ is not a solution to the equation:
		\[X^2+5X+6 = 0 \]
		The equation $X^2+5X+6 = 0$ has solutions $X=-2$ and $X=-3$.
	\end{example}
	
	\begin{definition}[Weak Solution of an Uncertain Equation]
		Let $f : \mathcal{X} \to \mathcal{Y}$ be an uncertain function, $C \in \tsbd{K}$. 
		A solution $X_0 \in \tsbd{K}$ is called a weak solution of the uncertain equation $f(X) = C$ if $f(X_0)$ is defined and $f(X_0) \subsetneq C$. If $f(X_0)$ and $C$ are finite-element uncertain numbers, we set: \[\gamma = \frac{\card(f(X_0))}{\card(C)} \in (0,1),\] and we write $(f(X_0) = C)_\gamma$, where the equality relation here is a weak equality relation. For example, if $\gamma = 0.5$, we write $\gtcl{f(X_0) = C}{0.5}$.
	\end{definition}
	
	\begin{theorem}[Extension of Weak Solutions to Exact Solutions]
		Let $f : \mathcal{X} \to \mathcal{Y}$ be an uncertain function and $C \in \tsbd{K}$. Suppose $f$ satisfies the monotonicity property under set inclusion (i.e., $X_1 \subseteq X_2 \implies f(X_1) \subseteq f(X_2)$). If $A \in \tsbd{K}$ is a weak solution of $f(X) = C$ with weak truth degree $(f(A) = C)_\gamma$ where $\gamma \in (0, 1)$, then there exists an exact solution $B \in \tsbd{K}$ of the equation $f(X) = C$ such that $A \subseteq B$.
	\end{theorem}
	
	\begin{proof}
		Let $A \in \tsbd{K}$ be a weak solution of $f(X) = C$ with weak truth degree $\gamma = \frac{\card(f(A))}{\card(C)} \in (0, 1)$. By definition of a weak solution, $f(A)$ is defined and satisfies $f(A) \subsetneq C$.
		
		To construct an exact solution $B \supseteq A$, consider the target complement set $C \setminus f(A) \neq \emptyset$. 
		
		For each element $c \in C \setminus f(A)$, since $c \in C \subseteq \mathcal{Y} = \text{Im}(f)$, there exists at least one element $x_c \in \mathcal{X}$ such that $c \in f(\{x_c\}_u)$. Construct the set $X_{\text{add}}$ by collecting one such preimage element $x_c$ for every $c \in C \setminus f(A)$:
		\[
		X_{\text{add}} = \s{x_c : c \in C \setminus f(A)}
		\]
		
		Define the expanded uncertain number $B \in \tsbd{K}$ as the set union of $A$ and $X_{\text{add}}$:
		\[
		B = A \cup X_{\text{add}}
		\]
		Clearly, $A \subseteq B$. By monotonicity of the uncertain function $f$ under set union:
		\[
		f(B) = f(A \cup X_{\text{add}}) = f(A) \cup f(X_{\text{add}})
		\]
		
		By construction of $X_{\text{add}}$, the image $f(X_{\text{add}})$ covers all remaining target values in $C \setminus f(A)$. Thus:
		\[
		f(B) = f(A) \cup (C \setminus f(A)) = C
		\]
		Since $f(B) = C$, the uncertain number $B$ is an exact solution of $f(X) = C$ containing $A$. This completes the proof.
	\end{proof}
	
	\section{Canonical Representation of Mappings and Equations}
	
	\begin{theorem}
		\label{dl:3.2}
		Let $X \in \mathcal{X}$ be an uncertain number with canonical form $X \sim (f_X, d_X)$ and $f: \mathcal{X} \rightarrow \mathcal{Y}$ be a single-variable uncertain mapping. 
		
		Then, the image $Y = f(X) \in \mathcal{Y}$ possesses a canonical form $Y \sim (f_Y, d_Y)$ determined in each arithmetic space as follows:
		
		\begin{enumerate}[leftmargin=*]
			\item On the Single-point Space $(\circ)_1$: 
			Since internal point identity is preserved, the scenario domain of $Y$ remains unchanged:
			\[
			d_Y = d_X
			\]
			The generating function $f_Y: d_Y \rightarrow \mathcal{Y}$ is defined via composition:
			\[
			\forall t \in d_Y, \quad f_Y(t) = f(f_X(t))
			\]
			
			\item On the Minkowski Space $(\circ)_m$: 
			If the expression of the function $f(X)$ contains $k$ occurrences of the variable $X$ ($k \in \mathbb{N}^*$), since each occurrence acts as an independent noise source, the scenario domain expands into a $k$-fold Cartesian product:
			\[
			d_Y = d_X^k = \underbrace{d_X \times d_X \times \dots \times d_X}_{k \text{ times}}
			\]
			The generating function $f_Y: d_Y \rightarrow \mathcal{Y}$ is defined by:
			\[
			\forall (t_1, t_2, \dots, t_k) \in d_Y, \quad f_Y(t_1, t_2, \dots, t_k) = f\left(f_X(t_1), f_X(t_2), \dots, f_X(t_k)\right)
			\]
		\end{enumerate}
	\end{theorem}
	\begin{proof}
		Let $X \in \mathcal{X}$ be an uncertain number with canonical form $X \sim (f_X, d_X)$, where $d_X \subseteq \mathbb{K}$ is the scenario index domain and $f_X: d_X \to \mathbb{K}$ is a surjective generating function satisfying $f_X(d_X) = X_s$. 
		
		We prove the canonical form $Y \sim (f_Y, d_Y)$ for the image $Y = f(X)$ in each arithmetic space separately:
		
		\begin{enumerate}[leftmargin=*]
			\item In Single-point Arithmetic Space $(\circ)_1$: \\
			By Definition 2.8, the single-point evaluation operator binds every occurrence of variable $X$ in the mapping $f$ to a single representative scenario value $x \in X_s$ during each evaluation pass. 
			
			Since $X \sim (f_X, d_X)$, every element $x \in X_s$ is generated by an index $t \in d_X$ via $x = f_X(t)$. Thus, internal point identity preserves the scenario domain without dimension expansion:
			\[
			d_Y = d_X
			\]
			The composite generating function $f_Y : d_Y \to \mathbb{K}$ defined by $f_Y(t) = f(f_X(t))$ for all $t \in d_Y$ maps $d_Y$ onto the scenario value set:
			\[
			f_Y(d_Y) = \{f(f_X(t)) \;:\; t \in d_X\} = \{f(x) \;:\; x \in X_s\} = \kgdd{f}(X)_s = Y_s
			\]
			Therefore, $Y \sim (f_Y, d_X)$ is the canonical form of $Y = \kgdd{f}(X)$ in $(\circ)_1$.
			
			\item In Minkowski Arithmetic Space $(\circ)_m$: \\
			By Definition 2.9, operations in Minkowski space model independent interactions among uncertain quantities. If the expression of $f(X)$ contains $k$ occurrences of variable $X$ ($k \in \mathbb{N}^*$), each occurrence acts as an independent noise source receiving an independent scenario value $x_i \in X_s$ for $i \in \{1, \dots, k\}$.
			
			Since each $x_i \in X_s$ is generated by an independent index $t_i \in d_X$ via $x_i = f_X(t_i)$, the overall scenario domain expands into the $k$-fold Cartesian product:
			\[
			d_Y = d_X^k = \underbrace{d_X \times d_X \times \dots \times d_X}_{k \text{ times}}
			\]
			The multi-variable generating function $f_Y : d_X^k \to \mathbb{K}$ defined by:
			\[
			f_Y(t_1, t_2, \dots, t_k) = f\left(f_X(t_1), f_X(t_2), \dots, f_X(t_k)\right), \quad \forall (t_1, \dots, t_k) \in d_X^k
			\]
			evaluates all independent combinations of scenario values across the Abstract Syntax Tree (AST) of $f(X)$. The image set under $f_Y$ satisfies:
			\begin{align*}
				f_Y(d_X^k) &= \{f\left(f_X(t_1), \dots, f_X(t_k)\right) \;:\; (t_1, \dots, t_k) \in d_X^k\} \\
				&= \{f(x_1, \dots, x_k) \;:\; x_1, \dots, x_k \in X_s\} = \kgdt{f}(X)_s = Y_s
			\end{align*}
			Therefore, $Y \sim (f_Y, d_X^k)$ is the canonical form of $Y = \kgdt{f}(X)$ in $(\circ)_m$.
		\end{enumerate}
		
		Both cases are verified, completing the proof.
	\end{proof}
	\begin{remark}
		Theorem \ref{dl:3.2} clarifies the essential boundary between the two arithmetic spaces: the concept of a ``single-variable function'' $f(X)$ in the single-point arithmetic space $(\circ)_1$ preserves the scenario dimension $d_Y = d_X$; whereas in the Minkowski space $(\circ)_m$, the algebraic expression of a single-variable function is deconstructed based on the $k$ occurrences of variable $X$, causing the scenario space to expand into $d_X^k$ corresponding to the Abstract Syntax Tree (AST).
	\end{remark}
	
	\begin{theorem}[Solution Characterization of Uncertain Equations]
		\label{thm:solution_characterization}
		Let $f : \mathcal{X} \to \mathcal{Y}$ be an uncertain function and $C \in \mathcal{Y}$ be a target uncertain constant with canonical representation $C \sim (f_C, d_C)$.
		
		Then, an element $X_0 \in \mathcal{X}$ is an exact solution of the equation $f(X) = C$ if and only if its image $F = f(X_0) \sim (f_F, d_F)$ satisfies the two-way scenario matching conditions:
		\begin{enumerate}[leftmargin=*]
			\item Image Inclusion ($f(X_0) \subseteq C$): 
			\[
			\forall t \in d_F, \quad \exists t' \in d_C \quad \text{such that} \quad f_F(t) = f_C(t')
			\]
			\item Target Coverage ($C \subseteq f(X_0)$): 
			\[
			\forall t' \in d_C, \quad \exists t \in d_F \quad \text{such that} \quad f_C(t') = f_F(t)
			\]
		\end{enumerate}
	\end{theorem}
	
	\begin{proof}
		By definition, $X_0 \in \mathcal{X}$ is an exact solution of the uncertain equation $f(X) = C$ if and only if $f(X_0)$ and $C$ share identical sets of scenario values, i.e., $F_s = C_s$.
		
		We prove both directions of the logical equivalence:
		
		\begin{enumerate}[leftmargin=*]
			\item Forward Direction ($\Rightarrow$): \\
			Suppose $X_0$ is an exact solution of $f(X) = C$, so $F_s = C_s$.
			\begin{itemize}[leftmargin=*]
				\item For any $t \in d_F$, the generated value $y = f_F(t) \in F_s$. Since $F_s = C_s$, we have $y \in C_s = f_C(d_C)$. By surjectivity of $f_C$ onto $C_s$, there exists $t' \in d_C$ such that $f_F(t) = f_C(t')$.
				\item Conversely, for any $t' \in d_C$, the generated value $c = f_C(t') \in C_s$. Since $C_s = F_s$, we have $c \in F_s = f_F(d_F)$. By surjectivity of $f_F$ onto $F_s$, there exists $t \in d_F$ such that $f_C(t') = f_F(t)$.
			\end{itemize}
			Thus, both matching conditions (1) and (2) hold.
			
			\item Backward Direction ($\Leftarrow$): \\
			Suppose both matching conditions (1) and (2) hold.
			\begin{itemize}[leftmargin=*]
				\item Condition (1) implies that for every $y \in F_s = f_F(d_F)$, there exists $t' \in d_C$ such that $y = f_C(t') \in C_s$. Thus $F_s \subseteq C_s$.
				\item Condition (2) implies that for every $c \in C_s = f_C(d_C)$, there exists $t \in d_F$ such that $c = f_F(t) \in F_s$. Thus $C_s \subseteq F_s$.
			\end{itemize}
			Combining both inclusions yields $F_s = C_s$, which means $f(X_0) = C$. Hence, $X_0$ is an exact solution of $f(X) = C$.
		\end{enumerate}
		
		This completes the proof.
	\end{proof}

			Thus, the scenario value set of $\hbd{c}{X}$ is defined by:
			\[
			(\hbd{c}{X})_s = \left\{ \sum_{k=1}^{\infty} \frac{x_k}{2^k} \;:\; x_k \in \{a, b\}, \; \forall k \in \mathbb{N}^* \right\}
			\]
			
			To prove $(\hbd{c}{X})_s = [a, b]$, we perform a change of variables. For each $k \in \mathbb{N}^*$, express $x_k \in \{a, b\}$ in terms of a binary choice variable $b_k \in \{0, 1\}$:
			\[
			x_k = a + (b - a)b_k, \quad \text{where } b_k = \begin{cases} 0 & \text{if } x_k = a \\ 1 & \text{if } x_k = b \end{cases}
			\]

			\begin{align*}
				\sum_{k=1}^{\infty} \frac{x_k}{2^k} &= \sum_{k=1}^{\infty} \frac{a + (b - a)b_k}{2^k} \\
				&= a \sum_{k=1}^{\infty} \frac{1}{2^k} + (b - a) \sum_{k=1}^{\infty} \frac{b_k}{2^k}
			\end{align*}
			
			Since $\sum_{k=1}^{\infty} \frac{1}{2^k} = 1$, the expression simplifies to:
			\[
			\sum_{k=1}^{\infty} \frac{x_k}{2^k} = a + (b - a) \sum_{k=1}^{\infty} \frac{b_k}{2^k}
			\]
			
			Notice that as $(b_1, b_2, b_3, \dots)$ ranges over all possible sequence combinations in $\{0, 1\}^{\mathbb{N}^*}$, the term $\sum_{k=1}^{\infty} \frac{b_k}{2^k}$ represents the standard base-2 (binary) fractional expansion of real numbers in the unit interval $[0, 1]$:
			\[
			\left\{ \sum_{k=1}^{\infty} \frac{b_k}{2^k} \;:\; b_k \in \{0, 1\}, \, \forall k \in \mathbb{N}^* \right\} = [0, 1]
			\]

			(\hbd{c}{X})_s = \{a + (b - a)t \;:\; t \in [0, 1]\} = [a, b]
			\]
			Converting back via the transfer operator yields $\hbd{c}{\{a, b\}_u} = [a, b]_u \equiv [a, b]$. This completes the proof.

	\section{Derivatives of Uncertain Functions}
	
	\begin{definition}[Difference Quotient]
		Let $f$ be an uncertain function from $\mathcal{X} \rightarrow \mathcal{Y}$ mapping $X \mapsto f(X)$, $h \in \mathbb{K}$.
		The difference quotient of the uncertain function $f(X)$ at $X_0$ is 
		\[\Delta_f(X_0)= \frac{f(X_0 + h) - f(X_0)}{h}\]
	\end{definition}
	
	\begin{definition}[Derivative]
		Let $f$ be an uncertain function from $\mathcal{X} \rightarrow \mathcal{Y}$ mapping $X \mapsto f(X)$.
		The derivative of the uncertain function $f(X)$ at $X_0$ is an uncertain number: 
		$$f'(X_0) := \s{\lim_{h\rightarrow 0}{a: a \in \Delta_f(X_0) , \lim_{h\rightarrow 0}a \in \mathbb{K}}}$$
	\end{definition}
	
	\begin{example}
		For $f(X) = C \in \tsbd{K}$, we have: $$f'(X) = 0$$
	\end{example}
	
	\begin{example}

	\begin{theorem}[Derivative of a Sum]
		Let $f : \mathcal{X} \to \mathcal{Y}, g : \mathcal{X} \to \mathcal{Y}$ be uncertain functions differentiable at $X_0 \in \tsbd{K}$. We have 
		\[
		((f+g)'(X_0))_\alpha = (f'(X_0) + g'(X_0))_\alpha, \quad \forall \alpha \in \set{1,1', m,m'}.
		\]
	\end{theorem}
	
	\section{Uncertain Integrals}
	
	\begin{definition}[Partial Uncertain Integral]
		Let $A,B \in \tsbd{K}$, $f: \mathcal{X} \to \mathcal{Y}$ be a function possessing an antiderivative, and $F$ be an antiderivative of $f$. The partial uncertain integral from $A$ to $B$ of $f(X)$ with respect to antiderivative $F$ is given by:
		\[
		\int_{A}^{B}{f(X)|F} := \s{k = \int_{a}^{b}f(x)dx : a \in F(H_A), b \in F(H_B), k \in \mathbb{K}}
		\]
		Where:
		\begin{align*}
			H_A &:= \left(\bigcup_{x \in A}{F^{-1}(x)}\right)_u \\
			H_B &:= \left(\bigcup_{x \in B}{F^{-1}(x)}\right)_u
		\end{align*}
	\end{definition}
	
	\begin{definition}[Uncertain Integral]
		Let $A,B \in \tsbd{K}$, $f: \tsbd{K} \to \tsbd{K}$ be a function possessing antiderivatives. Let $\mathcal{F}$ be the set of all antiderivatives of $f$. Then the uncertain integral from $A$ to $B$ of $f$ is given by 
		\[
		\int_{A}^{B}{f(X)} := \s{\int_{A}^{B}f(X)|F_i : F_i \in \mathcal{F}}
		\]
	\end{definition}
	
	\begin{example}
		Evaluate: 
		\[I = \int_{\s{1,2}}^{\s{3,4}}{(X+X)|(X^2)_{m'}}\]
		We have $F(X) = X*X$ as an antiderivative of $f(X)$. We compute:
		\begin{align*}
			F^{-1}(\s{1,2}) &= \s{-1,1,\sqrt{2},-\sqrt{2}} \\
			F^{-1}(\s{3,4}) &= \s{-2,2,\sqrt{3},-\sqrt{3}}
		\end{align*}
		Thus we have: 
		\begin{align*}
			F(F^{-1}(\s{1,2})) &= \s{-1,1,\sqrt{2},-\sqrt{2}} * \s{-1,1,\sqrt{2},-\sqrt{2}} \\
			F(F^{-1}(\s{3,4})) &= \s{-2,2,\sqrt{3},-\sqrt{3}} * \s{-2,2,\sqrt{3},-\sqrt{3}}
		\end{align*}
		Simplifying yields:
		\begin{align*}
			F(F^{-1}(\s{1,2})) &= \s{\pm 1, \pm 2, \pm \sqrt{2}} \\
			F(F^{-1}(\s{3,4})) &= \s{\pm 4, \pm 2\sqrt{3}, \pm 3}
		\end{align*}
		From which we calculate $I$ as equal to:
		$$
		\s{k = \int_{a}^{b}{2xdx}: a \in \s{\pm 1, \pm 2, \pm \sqrt{2}}, b \in \s{\pm 4, \pm 2\sqrt{3}, \pm 3}, k \in \mathbb{K}}
		$$
		Simplifying yields:
		\[I = \s{5, 7, 8, 10, 11, 12, 14, 15}\]
	\end{example}
	
	\begin{theorem}
		Let $A, B \in U(\mathbb{K})$ have canonical forms $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$. Let $g(X)$ be an uncertain function with structural antiderivative $F \in \mathcal{F}$ represented by the $k$-variable generating function $f_F(f_{X_1}, \dots, f_{X_k})$.
		
		Consider $k$ differential parameterization paths $f_{X_i}(t_i) = f_A(x_i) + t_i\big(f_B(y_i) - f_A(x_i)\big)$ with $t_i \in [0, 1]$. The total differential form of antiderivative $F$ is:
		\[
		d f_F = \sum_{i=1}^{k} \frac{\partial f_F}{\partial f_{X_i}} \cdot \big(f_B(y_i) - f_A(x_i)\big) dt_i
		\]
		The differential form uncertain integral from $A$ to $B$ of function $g(X)$ corresponding to antiderivative $F$, denoted $\int_A^B g(X)dX \Big| F$, has a canonical form generating function determined by the $k$-fold line integral:
		\[
		f_{I_F}(\mathbf{x}, \mathbf{y}) := \int_{[0, 1]^k} d f_F = f_F\big(f_B(y_1), \dots, f_B(y_k)\big) - f_F\big(f_A(x_1), \dots, f_A(x_k)\big)
		\]
		for all $(\mathbf{x}, \mathbf{y}) \in d_A^k \times d_B^k$.
	\end{theorem}
	\begin{proof}
		Let $A \sim (f_A, d_A)$ and $B \sim (f_B, d_B)$ in $\mathbf{U}(\mathbb{K})$. Under Minkowski arithmetic space $(\circ)_m$, the interaction between $A$ and $B$ via a $k$-variable structural antiderivative $F$ operates on the $2k$-fold Cartesian product index domain:
		\[
		d_{I_F} = d_A^k \times d_B^k = \underbrace{(d_A \times \dots \times d_A)}_{k \text{ times}} \times \underbrace{(d_B \times \dots \times d_B)}_{k \text{ times}}
		\]
		Fix an arbitrary index vector pair $(\mathbf{x}, \mathbf{y}) \in d_A^k \times d_B^k$, where $\mathbf{x} = (x_1, \dots, x_k) \in d_A^k$ and $\mathbf{y} = (y_1, \dots, y_k) \in d_B^k$.
		
		\begin{enumerate}[leftmargin=*]
			\item Parametric Path and Total Differential Formulation: \\
			For each component $i \in \{1, \dots, k\}$, define the linear parameterization path $\gamma_i : [0, 1] \to \mathbb{K}$ connecting $f_A(x_i)$ to $f_B(y_i)$:
			\[
			f_{X_i}(t_i) = (1 - t_i) f_A(x_i) + t_i f_B(y_i) = f_A(x_i) + t_i \Delta f_i, \quad t_i \in [0, 1]
			\]
			where $\Delta f_i = f_B(y_i) - f_A(x_i)$. The differential along each path is $d f_{X_i} = \Delta f_i \, dt_i$.
			
			By the multivariate chain rule, the total differential form $df_F$ along the hypercube path $\boldsymbol{\gamma}(\mathbf{t}) = (\gamma_1(t_1), \dots, \gamma_k(t_k))$ on $\mathbf{t} = (t_1, \dots, t_k) \in [0, 1]^k$ is:
			\[
			d f_F = \sum_{i=1}^{k} \frac{\partial f_F}{\partial f_{X_i}} d f_{X_i} = \sum_{i=1}^{k} \frac{\partial f_F}{\partial f_{X_i}} \big(f_B(y_i) - f_A(x_i)\big) dt_i
			\]
			
			\item Evaluation of the $k$-fold Line Integral: \\
			The generating function value $f_{I_F}(\mathbf{x}, \mathbf{y})$ is defined as the definite integral of the total differential form $df_F$ over the parameter space $[0, 1]^k$:
			\[
			f_{I_F}(\mathbf{x}, \mathbf{y}) = \int_{[0, 1]^k} d f_F
			\]
			Since $df_F$ is an exact total differential of the continuously differentiable $k$-variable function $f_F$, by the Fundamental Theorem of Calculus for line integrals in multiple variables (or gradient theorem), the integral depends solely on the boundary values of the path $\boldsymbol{\gamma}$ at $\mathbf{t} = \mathbf{1} = (1, \dots, 1)$ and $\mathbf{t} = \mathbf{0} = (0, \dots, 0)$:
			\begin{align*}
				\boldsymbol{\gamma}(\mathbf{1}) &= \big(f_{X_1}(1), \dots, f_{X_k}(1)\big) = \big(f_B(y_1), \dots, f_B(y_k)\big) \\
				\boldsymbol{\gamma}(\mathbf{0}) &= \big(f_{X_1}(0), \dots, f_{X_k}(0)\big) = \big(f_A(x_1), \dots, f_A(x_k)\big)
			\end{align*}
			Evaluating the exact differential integral yields:
			\begin{align*}
				\int_{[0, 1]^k} d f_F &= \int_{\boldsymbol{\gamma}(\mathbf{0})}^{\boldsymbol{\gamma}(\mathbf{1})} d f_F \\
				&= f_F\big(\boldsymbol{\gamma}(\mathbf{1})\big) - f_F\big(\boldsymbol{\gamma}(\mathbf{0})\big) \\
				&= f_F\big(f_B(y_1), \dots, f_B(y_k)\big) - f_F\big(f_A(x_1), \dots, f_A(x_k)\big)
			\end{align*}
			
			\item Canonical Representation of the Integral: \\
			Since $f_A(d_A) = A_s$ and $f_B(d_B) = B_s$, as $(\mathbf{x}, \mathbf{y})$ ranges over $d_A^k \times d_B^k$, the generating function $f_{I_F}$ maps $d_{I_F}$ onto the complete set of Minkowski evaluation values:
			\[
			f_{I_F}(d_A^k \times d_B^k) = \left\{ f_F(b_1, \dots, b_k) - f_F(a_1, \dots, a_k) \;:\; a_i \in A_s, \, b_i \in B_s, \, \forall i=1,\dots,k \right\}
			\]
			which corresponds exactly to the scenario value set of $\int_A^B g(X)dX \Big| F$ in Minkowski space $(\circ)_m$.
		\end{enumerate}
		
		Thus, $f_{I_F}(\mathbf{x}, \mathbf{y}) = f_F\big(f_B(y_1), \dots, f_B(y_k)\big) - f_F\big(f_A(x_1), \dots, f_A(x_k)\big)$ is the valid canonical generating function over $d_A^k \times d_B^k$. This completes the proof.
	\end{proof}
	\begin{theorem}
		Let $f,g : \mathcal{X} \to \mathcal{Y}$ be two uncertain functions differentiable at uncertain point $X_0 \in \mathcal{X}$. Suppose representative generating functions of $f(X_0)$ and $g(X_0)$ in canonical form are $f_{f(X_0)(t)}$ and $f_{g(X_0)}(t)$ respectively with $t \in [0,1]^k$.
		
		Then the differential form of the sum function:
		$h(x) = (f+g)(X)$ at $X_0$ is defined by the total linear differential operator:
		\[df_{f+g}(t) = df_{f(X_0)}(t) + df_{g(X_0)}(t)\]
		Explicit representation via partial derivatives over parameter hypercube $[0,1]^k:$
		\[df_{\left(f+g\right)}(t) = \sum_{i=1}^{k}\left[ \left(\frac{\partial f_{f(X_0)}(t)}{\partial t_i} + \frac{\partial f_{g(X_0)}(t)}{\partial t_i}\right) dt_i\right]\]
	\end{theorem}
	\begin{proof}
		Let $f, g : \mathcal{X} \to \mathcal{Y}$ be two uncertain functions differentiable at an uncertain point $X_0 \in \mathcal{X}$. Let canonical representations of $f(X_0)$ and $g(X_0)$ over the parameter hypercube $[0, 1]^k$ be given by generating functions $f_{f(X_0)}(t)$ and $f_{g(X_0)}(t)$, respectively, where $t = (t_1, \dots, t_k) \in [0, 1]^k$.
		
		\begin{enumerate}[leftmargin=*]
			\item Canonical Representation of the Sum Function: \\
			By the definition of the sum function $h(X) = (f + g)(X)$, for every parameter point $t \in [0, 1]^k$, the canonical generating function of $h(X_0) = (f+g)(X_0)$ is defined pointwise by:
			\[
			f_{(f+g)(X_0)}(t) = f_{f(X_0)}(t) + f_{g(X_0)}(t)
			\]
			
			\item Total Linear Differential Operator: \\
			Since $f(X_0)$ and $g(X_0)$ are differentiable at $X_0$, their generating functions $f_{f(X_0)}$ and $f_{g(X_0)}$ are continuously differentiable with respect to the parameter vector $t \in [0, 1]^k$. 
			
			Applying the total differential operator $d$ to $f_{(f+g)(X_0)}(t)$, by linearity of differentiation:
			\[
			d f_{(f+g)(X_0)}(t) = d \left[ f_{f(X_0)}(t) + f_{g(X_0)}(t) \right] = d f_{f(X_0)}(t) + d f_{g(X_0)}(t)
			\]
			
			\item Explicit Representation via Partial Derivatives: \\
			By definition of the total differential for a multivariate function over the hypercube $[0, 1]^k$:
			\begin{align*}
				d f_{f(X_0)}(t) &= \sum_{i=1}^{k} \frac{\partial f_{f(X_0)}(t)}{\partial t_i} dt_i \\
				d f_{g(X_0)}(t) &= \sum_{i=1}^{k} \frac{\partial f_{g(X_0)}(t)}{\partial t_i} dt_i
			\end{align*}
			Substituting these total differentials into the linear sum yields:
			\begin{align*}
				d f_{(f+g)(X_0)}(t) &= \sum_{i=1}^{k} \frac{\partial f_{f(X_0)}(t)}{\partial t_i} dt_i + \sum_{i=1}^{k} \frac{\partial f_{g(X_0)}(t)}{\partial t_i} dt_i \\
				&= \sum_{i=1}^{k} \left[ \left( \frac{\partial f_{f(X_0)}(t)}{\partial t_i} + \frac{\partial f_{g(X_0)}(t)}{\partial t_i} \right) dt_i \right]
			\end{align*}
		\end{enumerate}				
		This completes the proof.
	\end{proof}
