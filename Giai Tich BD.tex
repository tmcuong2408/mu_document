\begin{definition}[Giới hạn của một dãy số]
	Cho $(X_n)_{n \in \mathbb{N}}$ là một dãy các số bất định.
	Ta nói dãy số bất định $(X_n)$ hội tụ về số bất định $L$ nếu:
	Với mọi số thực $\epsilon > 0$, tồn tại số tự nhiên $n_0$ sao cho với mọi $n > n_0$, ta luôn có:$$d_H(X_n, L) < \epsilon$$
	Khi đó ta viết \[\lim_{n \to \infty} X_n = L\]
\end{definition}
\begin{definition}[Giới hạn của hàm số tại một điểm]
	Cho $f$ là một hàm bất định từ $\mathcal{X} \rightarrow \mathcal{Y}$ biến $X \mapsto f(X)$. Hàm bất định $f(X)$ được gọi là có giới hạn $L$ khi $X \rightarrow X_0$, ký hiệu là $$\lim\limits_{X \rightarrow X_0}f(X) = L$$
	Nếu với $$(\forall \epsilon >0)(\exists \delta >0)(\forall X \in \mathcal{X}) : $$ $$  d_H(X,X_0)<\delta \Rightarrow d_H(f(X),L) < \epsilon$$
\end{definition}
\begin{definition}[Hàm bất định liên tục]
	Hàm bất định $f : \mathcal{X} \to \mathcal{Y}$ được gọi là liên tục tại điểm $X_0 \in \mathcal{X}$ nếu như $$\lim\limits_{X \rightarrow X_0}f(X) = f(X_0)$$
\end{definition}
\begin{definition}[Tỷ sai phân]
	Cho $f$ là một hàm bất định từ $\mathcal{X} \rightarrow \mathcal{Y}$ biến $X \mapsto f(X)$, $h \in \mathbb{K}$.
	Tỷ sai phân của hàm bất định $f(X)$ tại $X_0$ là  \[\Delta_f(X_0)= \frac{f(X_0 + h) - f(X_0)}{h}\]
\end{definition}
\begin{definition}[Đạo hàm]
	Cho $f$ là một hàm bất định từ $\mathcal{X} \rightarrow \mathcal{Y}$ biến $X \mapsto f(X)$.
	Đạo hàm của hàm bất định của $f(X)$ tại $X_0$ là một số bất định: $$f'(X_0) := \s{\lim_{h\rightarrow 0}{a: a \in \Delta_f(X_0) , \lim_{h\rightarrow 0}a \in \mathbb{K}}}$$
\end{definition}
\begin{example}
	Cho $f(X) = C \in \tsbd{K}$ ta có: $$f'(X) = 0$$
\end{example}

	$$f'(X) = X^2_+ = X+X = \s{1,2} + \s{1,2} = \s{2,3,4}$$
\end{example}

\begin{theorem}[Đạo hàm của một tổng]
	Cho $f : \mathcal{X} \to \mathcal{Y}, g : \mathcal{X} \to \mathcal{Y}$ là những hàm bất định và có đạo hàm tại $X_0 \in \tsbd{K}$. Ta có \[
		((f+g)'(X_0))_\alpha = (f'(X_0) + g'(X_0))_\alpha, \forall \alpha \in \set{1,1',m,m'}.
	\]
\end{theorem}
\begin{definition}[Tích phân bất định riêng phần]
	Cho $A,B \in \tsbd{K}, f: \mathcal{X} \to \mathcal{Y}$ là một hàm số có nguyên hàm, $F$ là một nguyên hàm của $f$. Tích phân bất định riêng phần cận từ $A$ đến $B$ của $f(X)$ tại hàm $F$ cho bởi:
	\[
		\int_{A}^{B}{f(X)|F} := \s{k = \int_{a}^{b}f(x)dx : a \in F(H_A), b \in F(H_B), k \in \mathbb{K}}
	\]
	Trong đó:
	\begin{align*}
		H_A &:= \left(\bigcup_{x \in A}{F^{-1}(x)}\right)_u \\
		H_B &:= \left(\bigcup_{x \in B}{F^{-1}(x)}\right)_u
	\end{align*}
\end{definition}
\begin{definition}[Tích phân bất định]
	Cho $A,B \in \tsbd{K}, f: \tsbd{K} \to \tsbd{K}$ là một hàm số có nguyên hàm. Gọi $\mathcal{F}$ là tập tất cả các nguyên hàm của $f$. Khi đó tích phân bất định cận từ $A$ đến $B$ của $f$ cho bởi\[
	\int_{A}^{B}{f(X)} := \s{\int_{A}^{B}f(X)|F_i : F_i \in \mathcal{F}}
	\]
\end{definition}
\begin{example}
	Tính: \[I = \int_{\s{1,2}}^{\s{3,4}}{(X+X)|(X^2)_2}\]
	Ta có $F(X) = X*X$ là một nguyên hàm của $f(X)$. Ta tính:
	\begin{align*}
		F^{-1}(\s{1,2}) &= \s{-1,1,\sqrt{2},-\sqrt{2}} \\
		F^{-1}(\s{3,4}) &= \s{-2,2,\sqrt{3},-\sqrt{3}}
	\end{align*}
	Vậy ta có: 
	\begin{align*}
		F(F^{-1}(\s{1,2})) &= \s{-1,1,\sqrt{2},-\sqrt{2}} * \s{-1,1,\sqrt{2},-\sqrt{2}} \\
		F(F^{-1}(\s{3,4})) &= \s{-2,2,\sqrt{3},-\sqrt{3}} * \s{-2,2,\sqrt{3},-\sqrt{3}}
	\end{align*}
	Rút gọn ta được:
	\begin{align*}
		F(F^{-1}(\s{1,2})) &= \s{\pm 1, \pm 2, \pm \sqrt{2}} \\
		F(F^{-1}(\s{3,4})) &= \s{\pm 4, \pm 2\sqrt{3}, \pm 3}
	\end{align*}
	Từ đó ta tính được $I$ bằng:
	$$
		 \s{k = \int_{a}^{b}{2xdx}: a \in \s{\pm 1, \pm 2, \pm \sqrt{2}}, b \in  \s{\pm 4, \pm 2\sqrt{3}, \pm 3}, k \in \mathbb{K}}
	$$
	Rút gọn ta được:
	\[I = \s{5, 7, 8, 10, 11, 12, 14, 15}\]
\end{example}
\begin{theorem}
	Cho $A, B \in U(\mathbb{K})$ có dạng chuẩn tắc $A \sim (f_A, d_A)$ và $B \sim (f_B, d_B)$. Cho $g(X)$ là hàm bất định có nguyên hàm cấu trúc $F \in \mathcal{F}$ biểu diễn bởi hàm sinh $k$-biến $f_F(f_{X_1}, \dots, f_{X_k})$.
	
	Xét $k$ đường tham số hóa vi phân $f_{X_i}(t_i) = f_A(x_i) + t_i\big(f_B(y_i) - f_A(x_i)\big)$ với $t_i \in [0, 1]$. Dạng vi phân toàn phần của nguyên hàm $F$ là:
	\[
	d f_F = \sum_{i=1}^{k} \frac{\partial f_F}{\partial f_{X_i}} \cdot \big(f_B(y_i) - f_A(x_i)\big) dt_i
	\]
	Tích phân bất định dạng vi phân cận từ $A$ đến $B$ của hàm $g(X)$ ứng với nguyên hàm $F$, ký hiệu $\int_A^B g(X)dX \Big| F$, có hàm sinh dạng chuẩn tắc xác định bởi tích phân đường $k$-lớp:
	\[
	f_{I_F}(\mathbf{x}, \mathbf{y}) := \int_{[0, 1]^k} d f_F = f_F\big(f_B(y_1), \dots, f_B(y_k)\big) - f_F\big(f_A(x_1), \dots, f_A(x_k)\big)
	\]
	với mọi $(\mathbf{x}, \mathbf{y}) \in d_A^k \times d_B^k$.
\end{theorem}
