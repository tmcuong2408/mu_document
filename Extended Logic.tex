\chapter{Logic Mở Rộng}

\section{Không gian trạng thái và Mệnh đề nguyên tử}
Cho $(\Omega, \mathcal{F}, \lambda)$ là một không gian độ đo xác định đại diện cho tập hợp các kịch bản hoặc trạng thái khả dĩ của hệ thống.

\begin{definition}[Mệnh đề nguyên tử]
	Một mệnh đề nguyên tử $P$ trong hệ Logic Mở Rộng được định nghĩa gắn liền với một hàm chân lý kịch bản $v(P, \cdot) : \Omega \to \{0, 1\}$, trong đó:
	\[
	v(P, \omega) = \begin{cases} 
		1 & \text{nếu } P \text{ đúng tại trạng thái } \omega, \\
		0 & \text{nếu } P \text{ sai tại trạng thái } \omega.
	\end{cases}
	\]
\end{definition}

\begin{definition}[Toán tử lượng giá chân lý]
	Toán tử lượng giá chân lý $V$ là một ánh xạ từ tập hợp các mệnh đề vào đoạn $[0, 1]$, xác định bởi độ đo của tập các trạng thái làm cho mệnh đề nghiệm đúng:
	\[
	V(P) := \frac{\lambda(\{\omega \in \Omega : v(P, \omega) = 1\})}{\lambda(\Omega)}
	\]
\end{definition}

\section{Đại số mệnh đề phức hợp và Phép nối logic}
Các phép nối logic cổ điển (Phủ định $\neg$, Hội $\land$, Tuyển $\lor$, Kéo theo $\Rightarrow$) được mở rộng lên không gian trạng thái thông qua đại số Boole trên từng điểm $\omega \in \Omega$.

\begin{definition}[Lượng giá mệnh đề phức hợp]
	Cho $P_1, P_2, \dots, P_n$ là các mệnh đề nguyên tử và $\Phi$ là một mệnh đề phức hợp cấu thành từ các phép nối logic cổ điển. Giá trị chân lý tại trạng thái $\omega \in \Omega$ của $\Phi$ được xác định bởi:
	\[
	v(\Phi(P_1, \dots, P_n), \omega) = \Phi\big(v(P_1, \omega), v(P_2, \omega), \dots, v(P_n, \omega)\big) \in \{0, 1\}
	\]
	Khi đó, giá trị chân lý toàn cục $V(\Phi) \in [0, 1]$ được tính bởi:
	\[
	V(\Phi) := \frac{\lambda(\{\omega \in \Omega : v(\Phi(P_1, \dots, P_n), \omega) = 1\})}{\lambda(\Omega)}
	\]
\end{definition}

\begin{theorem}[Bảo toàn hằng đúng Logic cổ điển]
	Nếu $\Phi$ là một hằng đúng (tautology) của Logic mệnh đề cổ điển, thì $V(\Phi) = 1$. Đặc biệt, các quy luật nền tảng luôn được bảo toàn tuyệt đối:
	\begin{enumerate}
		\item Quy luật mâu thuẫn: $V(P \land \neg P) = 0$.
		\item Quy luật bài trung: $V(P \lor \neg P) = 1$.
	\end{enumerate}
\end{theorem}

\begin{proof}
	Vì $\Phi$ là hằng đúng cổ điển, với mọi phối hợp giá trị nhị phân của các mệnh đề nguyên tử, ta luôn có $\Phi\big(v(P_1, \omega), \dots, v(P_n, \omega)\big) = 1$ với mọi $\omega \in \Omega$.
	Do đó, $\{\omega \in \Omega : v(\Phi, \omega) = 1\} = \Omega$. Suy ra:
	\[
	V(\Phi) = \frac{\lambda(\Omega)}{\lambda(\Omega)} = 1
	\]
\end{proof}

\section{Quan hệ Kéo theo và Quy tắc Suy luận}

\begin{definition}[Phép kéo theo và Mức độ suy luận]
	Cho hai mệnh đề $P$ và $Q$. Phép kéo theo $P \Rightarrow Q$ tại trạng thái $\omega$ được xác định theo logic cổ điển $v(P \Rightarrow Q, \omega) = v(\neg P \lor Q, \omega)$. Giá trị chân lý toàn cục của phép kéo theo là:
	\[
	V(P \Rightarrow Q) = \frac{\lambda(\{\omega \in \Omega : v(P, \omega) \le v(Q, \omega)\})}{\lambda(\Omega)}
	\]
\end{definition}

\begin{theorem}[Bổ đề Modus Ponens Mở rộng]
	Với hai mệnh đề $P$ và $Q$ bất kỳ, ta luôn có bất đẳng thức suy luận:
	\[
	V(Q) \ge V(P) + V(P \Rightarrow Q) - 1
	\]
\end{theorem}

\section{Ví dụ minh họa tính bảo toàn tương quan}
\begin{example}
	Xét không gian trạng thái $\Omega = \{1, 2, 3, 4\}$ với độ đo đếm $\lambda(A) = |A|$. Xét hai mệnh đề nguyên tử $P$ và $Q$ có hàm chân lý kịch bản:
	\begin{itemize}
		\item $P$ đúng tại $\Omega_P = \{1, 2\} \implies V(P) = 2/4 = 0.5$.
		\item $Q$ đúng tại $\Omega_Q = \{2, 3\} \implies V(Q) = 2/4 = 0.5$.
	\end{itemize}
	Tập trạng thái làm cho mệnh đề phức hợp $P \land Q$ đúng là $\Omega_P \cap \Omega_Q = \{2\}$. Do đó:
	\[
	V(P \land Q) = \frac{|\{2\}|}{4} = 0.25
	\]
	\textit{Nhận xét:} Nếu dùng toán tử Fuzzy cổ điển $\min(V(P), V(Q)) = \min(0.5, 0.5) = 0.5$, kết quả sẽ bị sai lệch do bỏ qua sự lệch kịch bản giữa $P$ và $Q$. Việc lượng giá trực tiếp trên không gian trạng thái $\Omega$ giúp giữ nguyên bản chất tương quan logic.
\end{example}
