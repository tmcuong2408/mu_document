\section{Một số khái niệm}
\begin{definition}[Hàm bất định]
	Cho $\mathcal{X},\mathcal{Y}$ là các tập con của $\tsbd{K}$,  một ánh xạ $f : \mathcal{X} \to \mathcal{Y}$ được gọi là một hàm bất định.
\end{definition}
\begin{definition}[Hàm bất định đơn điểm]
	Cho tập bất định $A \in \tsbd{K}$. Ta gọi \textbf{hàm bất định đơn điểm} là một ánh xạ $f$ tác động lên biến $A$, sao cho tại mỗi lượt ánh xạ, biến $A$ chỉ nhận duy nhất một giá trị điểm $x \in A$ đại diện tại tất cả các vị trí xuất hiện của $A$ trong biểu thức $f$.
	
	Tập giá trị của hàm bất định theo điểm tương ứng, ký hiệu là $\kgdd{f}$, được xác định bởi:
	\[
	\kgdd{f}(A) := \s{f(x) \;:\; x \in A}
	\]
	Và gọi các phép toán nằm trong $(\circ)_1$ là các phép toán trên không gian số học số 1.
\end{definition}
\begin{example}
	Ví dụ này để độc giả tránh nhầm lẫn các không gian số học trong nghiên cứu này, ta nêu các bài toán sau:
	\begin{enumerate}
		\item Anh $A$ định xây một cái sân hình vuông, chiều dài cạnh thuộc tập $\set{1,2}$. Vì sau khi chọn được 1 cạnh thì cạnh còn lại có chiều dài cũng phải bằng với con số đã chọn đó nên ta có diện tích cái sân là: \[S_1 = \kgdd{\s{1,2}^2} = \s{1,4}\]
		\item Anh $B$ định xây một cái sân nhưng $B$ giao hai người quyết định chiều dài các cạnh, mỗi người có sự lựa chọn chiều dài một cạnh thuộc tập $\set{1,2}$, khi đó diện tích sân của anh $B$ là:
		\[S_2 = \kgdt{\s{1,2}^2} =\s{1,2,4}\]
	\end{enumerate}
	Vậy tổng diện tích sân nhà anh $A$ và anh $B$ là:\[S= S_1 + S_2 = \s{2, 3, 5, 6, 8}\]
	Gọi $X \in \tsbd{K}$ là số bất định chỉ lựa chọn khả dĩ để xây sân của nhà anh $A$ và anh $B$, hàm tổng diện tích sân nhà anh $A$ và anh $B$ sẽ bằng:
	\[f(X) = \kgdd{X^2} + \kgdtmr{X^2}\] là một hàm bất định.
\end{example}
\begin{example}
	Cho $f : \tsbd{K} \to \tsbd{K}$ được cho bởi:
	\[f(X) = (X^2+5X+6)_1\] là một hàm bất định đơn điểm.
\end{example}
\begin{remark}
	Định nghĩa này đảm bảo tính nhất quán về mặt đại số nhờ hai cơ chế cốt lõi:
	\begin{itemize}
		\item \textbf{Tính đồng nhất điểm nội bộ:} Mọi ký hiệu biến $A$ xuất hiện trong biểu thức $f$ (dù ở các vị trí hay phép toán khác nhau) đều bị ràng buộc nhận chung một giá trị $x \in A$ trong cùng một lượt ánh xạ.
		\item \textbf{Triệt tiêu hiện tượng phình đại lượng:} Nhờ cơ chế gán điểm đồng nhất, các tính chất đại số cổ điển được bảo toàn trọn vẹn, chẳng hạn như $\kgdd{x - x}(A) = \s{0}$ và $\kgdd{x + x}(A) = 2A$.
	\end{itemize}
\end{remark}

\begin{example}
	Xét hàm một biến $f(A) = A^2 - 3A + 2$ với $A \in \tsbd{\mathbb{R}}$. Tập giá trị của hàm bất định đơn điểm này được xác định bởi:
	\[
	\kgdd{f}(A) = \s{x^2 - 3x + 2 \;:\; x \in A}
	\]
	Tại mỗi lượt quét $x \in A$, cả vị trí $A^2$ và $-3A$ đều đồng nhất nhận cùng một giá trị $x$.
\end{example}
\begin{definition}
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định trên không gian số học 1, $X \in \mathcal{X}$. Ta định nghĩa phép toán ngược của $f$ trên không gian số học 1 mở rộng, ký hiệu là $\kgddmr{\circ}$:
	\[
		\kgddmr{f^{-1}(X)} := \bd{\bigcup_{x \in X}{f^{-1}(x)}}
	\]
	Khi đó trên không gian số học 1 ta luôn có: \begin{enumerate}
		\item $f(f^{-1}(X)) \subseteq X$.
		\item Nếu với mọi $x \in X$ ta luôn có $f^{-1}(x) \neq \emptyset$ thì $f(f^{-1}(X)) = X$.
	\end{enumerate} 
\end{definition}
\begin{example}
	\begin{enumerate}
		\item 	Cho $f(X)=\kgdd{X^2}$, ta định nghĩa: \[\kgddmr{\sqrt{X}} = \bd{\bigcup_{x \in X}{\set{a \in K : a^2 = x}}}\]
		Dễ thấy nếu mọi $x \in X$ đều tồn tại $a \in \mathbb{K}$ thỏa $a^2=x$ thì ta có: \[\kgdd{\left(\sqrt{X}\right)^2} = X \]
		\item Cho $f(X)=\kgdd{\sin{X}}$, ta định nghĩa: \[\kgddmr{\arcsin{X}} = \bd{\bigcup_{x \in X}{\set{a \in \mathbb{K} : sin(a) = x}}}\]
	\end{enumerate}
\end{example}
\begin{definition}
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định trên không gian số học Minkowski mở rộng,ký hiệu là $\kgdtmr{\circ}$, $X \in \mathcal{X}$. Ta định nghĩa phép toán ngược của $f$ trên không gian số học Minkowski mở rộng:
	\[
	\kgdtmr{f^{-1}(X)} := \s{A \in \tsbd{K}: f(A) = X}
	\]
	Khi đó ta có : $\left[f\left(\kgdtmr{f^{-1}(X)}\right)\right]_1\subseteq X$
\end{definition}
\begin{example}
Cho $f(X)=\kgdt{X^2}$, ta định nghĩa: \[\kgdtmr{\sqrt{X}} = \s{A \in \tsbd{K}: A^2 = X}\]
Chẳng hạn \[\kgdtmr{\sqrt{\s{1,2,4}}} = \s{\s{1,2},\s{-1,-2}}\]
\end{example}
\begin{definition}[Đồ thị của hàm bất định tại một điểm]
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $A \in \mathcal{X}$ ta định nghĩa đồ thị của hàm bất định $f$ tại điểm $A$ là:
	\[pgr_f(A):= A \times f(A)\]
\end{definition}
\begin{definition}[Đồ thị của hàm bất định]
	Cho $\mathcal{X} ,\mathcal{Y}$ là các tập con của $\tsbd{K}$ với $\mathbb{K}$ là một trường tùy ý.
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $A \in \mathcal{X}$ ta định nghĩa đồ thị của hàm bất định $f$ tại điểm $A$ là:
	\[\gr{f}:= \{A \times f(A) : A, f(A) \in \tsbd{K}\}\]
\end{definition}
\begin{example}
	Cho $f = X+X$ ta có \[\pgr_f{\s{1,2}} = \s{(1,2),(1,3),(1,4),(2,2),(2,3),(2,4)}\]
\end{example}

\begin{definition}[Vết]
	Cho $f$ là một hàm bất định từ $\mathcal{X}$ vào $\mathcal{Y}$ biến $X  \mapsto Y = f(X)$.
	\newline \noindent
	Xét $y \in  Y = f(X)$ ta định nghĩa vết của $y$ trong biểu thức của $f(X)$ là $\Tr(y \in f(X))$ là tập tất cả các biểu thức hình thức mà kết quả là bằng $y$ có được bằng cách tính $f(X)$.
\end{definition}
\begin{definition}[Phương trình bất định]
	Cho $f : \mathcal{X}\to  \mathcal{Y}$ là một hàm bất định, $C$ là một hằng số bất định, phương trình bất định ẩn $X$ là phương trình: \[f(X) = C\]
\end{definition}
\begin{example}
	Cho phương trình bất định \[\kgdd{X^2+5X+6}  = 0 \]
	Dễ thấy $X = \s{-2,-3}$ là một nghiệm của phương trình này. \\
	Nhưng $X = \s{-2,-3}$ không phải là nghiệm của phương trình:
	\[X^2+5X+6  = 0 \]
	Phương trình $X^2+5X+6  = 0$ có nghiệm là $X=-2$ và $X=-3$
\end{example}
\begin{definition}[Nghiệm yếu của phương trình bất định]
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $C \in \tsbd{K}$. 
	Nghiệm $X_0 \in \tsbd{K}$  được gọi là nghiệm yếu của phương trình bất định $f(X) = C$ nếu như $f(X_0)$ xác định và $f(X_0) \subsetneq C$. Nếu $f(X_0)$ và $C$ là số bất định hữu hạn phần tử thì ta đặt: \[\gamma = \frac{\card(f(X_0))}{\card(C)} \in (0,1),\] và ta viết $(f(X_0) = C)_\gamma$ với quan hệ bằng ở đây là quan hệ bằng yếu. Chẳng hạn nếu như $\gamma = 0.5$ thì ta viết $\gtcl{f(X_0) = C}{0.5}$.
\end{definition}
\begin{theorem}
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $C \in \tsbd{K}$. Nếu phương trình bất định $f(X) = C$ có nghiệm thì tồn tại  $A,B \in \tsbd{K} $ sao cho $\gtcl{f(A) = C}{\gamma} $ và $B$ là nghiệm của $f(X)=C$  và  $A \subseteq B$.
\end{theorem}
\par{Ta gọi phép toán được nằm trong ký hiệu $\kgdtmr{\circ}$ là các phép toán trên không gian số học số Minkowski mở rộng. Một trong số chúng được định nghĩa như sau:}
\begin{definition}
	Cho $p,q \in \mathbb{N}^*$
	Ta định nghĩa \[\kgdtmr{\frac{p}{q}A} := \s{X: \kgdt{X^q_+} = \kgdt{A^p_+}} \]
		Nếu $q = 1$ thì ta đặt:  \[\kgdtmr{pA}:= \kgdtmr{\frac{p}{1} A} \]
\end{definition}
\begin{example}
	$\kgdtmr{\frac{1}{2} \s{2,3,4}} = \s{\s{1,2}} = \s{1,2}$
\end{example}
\begin{definition}
	Cho $p,q \in \mathbb{N}$
	Ta định nghĩa \[\kgdtmr{A^{\frac{p}{q}}} := \s{X: \kgdt{X^q_*} = \kgdt{A^p_*}} \]
	Nếu $p = 1$  thì ta đặt: \[\kgdtmr{\sqrt[q]{A}} := \kgdtmr{A^{\frac{1}{q}}}\]
\end{definition}
