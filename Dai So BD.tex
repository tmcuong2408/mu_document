\section{Một số khái niệm}
\begin{definition}[Hàm bất định]
	Cho $\mathcal{X},\mathcal{Y}$ là các tập con của $\tsbd{K}$,  một ánh xạ $f : \mathcal{X} \to \mathcal{Y}$ được gọi là một hàm bất định.
\end{definition}
\begin{definition}
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định trên không gian số học $\alpha$ với $\alpha \in \set{1,m,m'}$, $X \in \mathcal{X}$. Ta định nghĩa phép toán ngược của $f$ trên không gian số học $\alpha$:
	\[
		(f^{-1}(X))_{\alpha} := \s{A \in \tsbd{K}: (f(A))_\alpha = X}
	\]
\end{definition}
\begin{example}
	\begin{enumerate}
		\item 	Cho $f(X)=\kgdd{X^2}$, ta định nghĩa: \[\kgdd{\sqrt{X}} = \s{A \in \tsbd{K} : \kgdd{X^2} = X}\]
		\item Cho $f(X)=\kgdd{\sin{X}}$, ta định nghĩa: \[\kgdd{\arcsin{X}} = \s{A \in \tsbd{K} : \kgdd{sin(A)} = X}\]
	\end{enumerate}
\end{example}
\begin{example}
Cho $f(X)=X^{\s{1,2,5}}$, ta định nghĩa: \[\sqrt[\s{1,2,5}]{X} = \s{A \in \tsbd{K} : A^{\s{1,2,5}} = X}\]
Chẳng hạn \[\sqrt[\s{1,2,5}]{\s{1, 2, 3, 4, 9, 32, 243}} = \s{1,2,3}\]
\end{example}
\begin{example}
	Cho $f(X)=\kgdtmr{X^2}$, ta định nghĩa: \[\kgdtmr{\sqrt{X}} = \s{A \in \tsbd{K} : \kgdtmr{A^2} = X}\]
	Chẳng hạn \[\kgdtmr{\sqrt{\s{1, 2, 4}}} = \s{\s{1,2},\s{-1,-2}}\]
\end{example}
\begin{definition}[Đồ thị của hàm bất định tại một điểm]
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $A \in \mathcal{X}$ ta định nghĩa đồ thị của hàm bất định $f$ tại điểm $A$ là:
	\[pgr_f(A):= A \times f(A)\]
\end{definition}
\begin{definition}[Đồ thị của hàm bất định]
	Cho $\mathcal{X} ,\mathcal{Y}$ là các tập con của $\tsbd{K}$ với $\mathbb{K}$ là một trường tùy ý.
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $A \in \mathcal{X}$ ta định nghĩa đồ thị của hàm bất định $f$ tại điểm $A$ là:
	\[\gr{f}:= \set{\pgr_f(X) : X \in \mathcal{X}}\]
\end{definition}
\begin{example}
	Cho $f = X+X$ ta có \[\pgr_f{\s{1,2}} = \s{(1,2),(1,3),(1,4),(2,2),(2,3),(2,4)}\]
\end{example}

\begin{definition}[Vết]
	Cho $f$ là một hàm bất định từ $\mathcal{X}$ vào $\mathcal{Y}$ biến $X  \mapsto Y = f(X)$.
	\newline \noindent
	Xét $y \in  Y = f(X)$ ta định nghĩa vết của $y$ trong biểu thức của $f(X)$ là $\Tr(y \in f(X))$ là tập tất cả các biểu thức hình thức mà kết quả là bằng $y$ có được bằng cách tính $f(X)$.
\end{definition}
\begin{definition}[Phương trình bất định]
	Cho $f : \mathcal{X}\to  \mathcal{Y}$ là một hàm bất định, $C$ là một hằng số bất định, phương trình bất định ẩn $X$ là phương trình: \[f(X) = C\]
\end{definition}
\begin{example}
	Cho phương trình bất định \[\kgdd{X^2+5X+6}  = 0 \]
	Dễ thấy $X = \s{-2,-3}$ là một nghiệm của phương trình này. \\
	Nhưng $X = \s{-2,-3}$ không phải là nghiệm của phương trình:
	\[X^2+5X+6  = 0 \]
	Phương trình $X^2+5X+6  = 0$ có nghiệm là $X=-2$ và $X=-3$
\end{example}
\begin{definition}[Nghiệm yếu của phương trình bất định]
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $C \in \tsbd{K}$. 
	Nghiệm $X_0 \in \tsbd{K}$  được gọi là nghiệm yếu của phương trình bất định $f(X) = C$ nếu như $f(X_0)$ xác định và $f(X_0) \subsetneq C$. Nếu $f(X_0)$ và $C$ là số bất định hữu hạn phần tử thì ta đặt: \[\gamma = \frac{\card(f(X_0))}{\card(C)} \in (0,1),\] và ta viết $(f(X_0) = C)_\gamma$ với quan hệ bằng ở đây là quan hệ bằng yếu. Chẳng hạn nếu như $\gamma = 0.5$ thì ta viết $\gtcl{f(X_0) = C}{0.5}$.
\end{definition}
\begin{theorem}
	Cho $f : \mathcal{X} \to \mathcal{Y}$ là một hàm bất định, $C \in \tsbd{K}$. Nếu phương trình bất định $f(X) = C$ có nghiệm thì tồn tại  $A,B \in \tsbd{K} $ sao cho $\gtcl{f(A) = C}{\gamma} $ và $B$ là nghiệm của $f(X)=C$  và  $A \subseteq B$.
\end{theorem}
